local function deleteHiddenScripts()
    local deletedCount = 0
    for _, obj in pairs(game:GetDescendants()) do
        if (obj:IsA("Script") or obj:IsA("LocalScript") or obj:IsA("ModuleScript")) then
            for _, name in pairs(scriptsToDelete) do
                if obj.Name == name then
                    pcall(function()
                        obj:Destroy()
                        deletedCount = deletedCount + 1
                        print("Deleted: " .. obj.Name .. " from " .. tostring(obj.Parent))
                    end)
                end
            end
        end
    end
    if getnilinstances then
        for _, obj in pairs(getnilinstances()) do
            if obj:IsA("LuaSourceContainer") then
                for _, name in pairs(scriptsToDelete) do
                    if obj.Name == name then
                        pcall(function()
                            obj:Destroy()
                            deletedCount = deletedCount + 1
                            print("Deleted nil script: " .. obj.Name)
                        end)
                    end
                end
            end
        end
    end
    return deletedCount
end

local function checkAndDestroy(script)
    if script:IsA("LocalScript") or script:IsA("Script") then
        local scriptName = script.Name
        local scriptNameLower = scriptName:lower()

        if scriptName == "" or scriptName:match("^%s*$") then
            warn("Destroying script with empty/whitespace name:", script:GetFullName())
            pcall(function()
                script.Disabled = true
                script:Destroy()
            end)
            return true
        end

        for _, keyword in pairs(blockedKeywords) do
            if scriptNameLower:find(keyword:lower()) then
                warn("Destroying blocked script:", scriptName, "Path:", script:GetFullName())
                pcall(function()
                    script.Disabled = true
                    script:Destroy()
                end)
                return true
            end
        end
    end
    return false
end

local function FindAllBalls()
    TrackedBalls = {}
    local tps = workspace:FindFirstChild("TPSSystem")
    if tps then
        local tpsBall = tps:FindFirstChild("TPS")
        if tpsBall and tpsBall:IsA("BasePart") then table.insert(TrackedBalls, tpsBall) end
    end
    local practice = workspace:FindFirstChild("Practice")
    if practice then
        for _, child in ipairs(practice:GetChildren()) do
            if child:IsA("BasePart") and (child.Name == "PSoccerBall" or child.Name == "Ball") then
                table.insert(TrackedBalls, child)
            end
        end
    end
    for _, n in ipairs(BALL_NAMES) do
        local b = workspace:FindFirstChild(n)
        if b and b:IsA("BasePart") and not table.find(TrackedBalls, b) then
            table.insert(TrackedBalls, b)
        end
    end
end

local function GetMainBall()
    local tps = workspace:FindFirstChild("TPSSystem")
    if tps then
        local b = tps:FindFirstChild("TPS")

    end
    for _, name in ipairs(BALL_NAMES) do
        local ball = workspace:FindFirstChild(name)
        if ball and ball:IsA("BasePart") then return ball end
    end
    return nil
end

local LimbTab = Window:Tab({ Title = "Limb Reach", Icon = "crosshair" })

local reachSettings = {
    HeadEnabled = false, HeadVisualizer = false, HeadVisTransparency = 0.7,
    ReachX_Head = 1, ReachY_Head = 1, ReachZ_Head = 1,
    RightArmEnabled = false, RightArmVisualizer = false, ArmVisTransparency = 0.7,
    LeftArmEnabled = false, LeftArmVisualizer = false,
    ReachX_Arm = 1, ReachY_Arm = 1, ReachZ_Arm = 1,
    RightLegEnabled = false, RightLegVisualizer = false, LegVisTransparency = 0.7,
    LeftLegEnabled = false, LeftLegVisualizer = false,
    ReachX_Leg = 1, ReachY_Leg = 1, ReachZ_Leg = 1,
    ReachDistance = 50,
}
local LimbHitboxes = {}
local LimbVisualizers = {}
local LimbActiveTouches = {}
local LIMB_TOUCH_COOLDOWN = 0.1
local LimbBodyParts = {"Head", "Right Arm", "Left Arm", "Right Leg", "Left Leg"}

local function SafeClear(tbl)
    if not tbl then return end
    if table.clear then table.clear(tbl) else for k in pairs(tbl) do tbl[k] = nil end end
end

local function GetReachDistanceForPart(partName)
    if partName == "Head" then
        if not reachSettings.HeadEnabled then return 0 end
        local avg = (reachSettings.ReachX_Head + reachSettings.ReachY_Head + reachSettings.ReachZ_Head) / 3
        return reachSettings.ReachDistance * avg
    elseif partName == "Right Arm" or partName == "Left Arm" then
        local side = partName:match("(%w+)%s")
        if not reachSettings[side .. "ArmEnabled"] then return 0 end
        local avg = (reachSettings.ReachX_Arm + reachSettings.ReachY_Arm + reachSettings.ReachZ_Arm) / 3
        return reachSettings.ReachDistance * avg
    elseif partName == "Right Leg" or partName == "Left Leg" then
        local side = partName:match("(%w+)%s")
        if not reachSettings[side .. "LegEnabled"] then return 0 end
        local avg = (reachSettings.ReachX_Leg + reachSettings.ReachY_Leg + reachSettings.ReachZ_Leg) / 3
        return reachSettings.ReachDistance * avg
    end
    return 0
end

local function CreateLimbHitbox(bodyPart, partName)
    local hitbox = Instance.new("Part")
    hitbox.Name = partName .. "_Hitbox"
    hitbox.Anchored = false
    hitbox.CanCollide = false
    hitbox.Transparency = 1
    hitbox.Massless = true
    if partName == "Head" then
        hitbox.Size = Vector3.new(
            bodyPart.Size.X * reachSettings.ReachX_Head * 2,
            bodyPart.Size.Y * reachSettings.ReachY_Head * 2,
            bodyPart.Size.Z * reachSettings.ReachZ_Head * 2
        )
    elseif partName == "Right Arm" or partName == "Left Arm" then
        hitbox.Size = Vector3.new(
            bodyPart.Size.X * reachSettings.ReachX_Arm * 2,
            bodyPart.Size.Y * reachSettings.ReachY_Arm * 2,
            bodyPart.Size.Z * reachSettings.ReachZ_Arm * 2
        )
    else
        hitbox.Size = Vector3.new(
            bodyPart.Size.X * reachSettings.ReachX_Leg * 2,
            bodyPart.Size.Y * reachSettings.ReachY_Leg * 2,
            bodyPart.Size.Z * reachSettings.ReachZ_Leg * 2
        )
    end
    hitbox.CFrame = bodyPart.CFrame
    hitbox.Parent = bodyPart
    local weld = Instance.new("WeldConstraint")
    weld.Part0 = bodyPart
    weld.Part1 = hitbox
    weld.Parent = hitbox
    return hitbox
end

local function GetVisTransparencyFor(partName)
    if partName == "Head" then return reachSettings.HeadVisTransparency
    elseif partName == "Right Arm" or partName == "Left Arm" then return reachSettings.ArmVisTransparency
    else return reachSettings.LegVisTransparency end
end

local function UpdateLimbVisualizer(partName)
    local char = player.Character
    if not char then return end
    local bodyPart = char:FindFirstChild(partName)
    if not bodyPart then return end
    local reachDist = GetReachDistanceForPart(partName)
    if reachDist > 0 then
        if LimbHitboxes[partName] then LimbHitboxes[partName]:Destroy() end
        LimbHitboxes[partName] = CreateLimbHitbox(bodyPart, partName)
    else
        if LimbHitboxes[partName] then LimbHitboxes[partName]:Destroy() LimbHitboxes[partName] = nil end
    end
    local visOn = false
    if partName == "Head" then visOn = reachSettings.HeadVisualizer
    elseif partName == "Right Arm" then visOn = reachSettings.RightArmVisualizer
    elseif partName == "Left Arm" then visOn = reachSettings.LeftArmVisualizer
    elseif partName == "Right Leg" then visOn = reachSettings.RightLegVisualizer
    elseif partName == "Left Leg" then visOn = reachSettings.LeftLegVisualizer end
    if LimbVisualizers[partName] then LimbVisualizers[partName]:Destroy() LimbVisualizers[partName] = nil end
    if visOn and LimbHitboxes[partName] then
        local hitbox = LimbHitboxes[partName]
        local vis = Instance.new("Part")
        vis.Name = partName .. "_Visualizer"
        vis.Size = hitbox.Size
        vis.CFrame = hitbox.CFrame
        vis.Anchored = false
        vis.CanCollide = false
        vis.Transparency = GetVisTransparencyFor(partName)
        vis.Color = Color3.fromRGB(0, 255, 100)
        vis.Material = Enum.Material.Neon
        vis.Massless = true
        vis.Parent = hitbox
        local w = Instance.new("WeldConstraint")
        w.Part0 = hitbox
        w.Part1 = vis
        w.Parent = vis
        LimbVisualizers[partName] = vis
    end
end

local function RefreshVisTransparency()
    for partName, v in pairs(LimbVisualizers) do
        if v and v.Parent then v.Transparency = GetVisTransparencyFor(partName) end
    end
end

local function FindBallPartLimb(part)
    if part:IsA("BasePart") and (part.Name == "TPS" or part.Name == "PSoccerBall" or BALL_SET[part.Name]) then return part end
    local current = part
    while current and current ~= Workspace do
        if current:IsA("BasePart") and (current.Name == "PSoccerBall" or current.Name == "TPS" or BALL_SET[current.Name]) then return current end
        current = current.Parent
    end
    return nil
end

local function CreateAllLimbHitboxes()
    local char = player.Character
    if not char then return end
    for _, partName in ipairs(LimbBodyParts) do
        local bodyPart = char:FindFirstChild(partName)
        if bodyPart and GetReachDistanceForPart(partName) > 0 then
            if LimbHitboxes[partName] then LimbHitboxes[partName]:Destroy() end
            LimbHitboxes[partName] = CreateLimbHitbox(bodyPart, partName)
            UpdateLimbVisualizer(partName)
        end
    end
end

local function RemoveAllLimbHitboxes()
    for _, h in pairs(LimbHitboxes) do if h then h:Destroy() end end
    for _, v in pairs(LimbVisualizers) do if v then v:Destroy() end end
    SafeClear(LimbHitboxes)
    SafeClear(LimbVisualizers)
    SafeClear(LimbActiveTouches)
end

local function CheckLimbBallCollisions()
    local char = player.Character
    if not char or not char.Parent then return end
    local currentTime = tick()
    local newTouches = {}
    for partName, hitbox in pairs(LimbHitboxes) do
        if hitbox and hitbox.Parent then
            local bodyPart = char:FindFirstChild(partName)
            if bodyPart then
                local op = OverlapParams.new()
                op.FilterDescendantsInstances = {char}
                op.FilterType = Enum.RaycastFilterType.Exclude
                local partsInBox = Workspace:GetPartBoundsInBox(hitbox.CFrame, hitbox.Size, op)
                for _, part in ipairs(partsInBox) do
                    local ballPart = FindBallPartLimb(part)
                    if ballPart then
                        local key = partName .. ":" .. ballPart:GetFullName()
                        newTouches[key] = {part = ballPart, bodyPart = bodyPart, time = currentTime}
                        if not LimbActiveTouches[key] or (currentTime - LimbActiveTouches[key].time) > LIMB_TOUCH_COOLDOWN then
                            LimbActiveTouches[key] = {part = ballPart, bodyPart = bodyPart, time = currentTime}
                            pcall(function()
                                firetouchinterest(ballPart, bodyPart, 0)
                                task.wait()
                                firetouchinterest(ballPart, bodyPart, 1)
                            end)
                        end
                    end
                end
            end
        end
    end
    for key, d in pairs(LimbActiveTouches) do
        if not newTouches[key] and (currentTime - d.time) > LIMB_TOUCH_COOLDOWN then LimbActiveTouches[key] = nil end
    end
end

pcall(CreateAllLimbHitboxes)

local limbLastUpdate = 0
RunService.Heartbeat:Connect(function()
    local now = tick()
    if now - limbLastUpdate >= 0.1 then
        limbLastUpdate = now
        pcall(CheckLimbBallCollisions)
    end
end)

            local mt = getrawmetatable(game)
            setreadonly(mt, false)
            local old_namecall = mt.__namecall
            local intercepted, disabled, returns = {}, {}, {}
            local namecallMethod = getnamecallmethod or get_namecall_method
            local methods = {
                RemoteEvent = "FireServer",
                RemoteFunction = "InvokeServer",
                BindableEvent = "Fire",
                BindableFunction = "Invoke"
            }

            getgenv().InterceptRemoteArgs = function(eventOrFunc, args)
                intercepted[eventOrFunc] = args
            end

            getgenv().SpoofReturn = function(func, newReturn)
                returns[func] = newReturn
            end

            getgenv().DisableRemote = function(eventOrFunc)
                disabled[eventOrFunc] = true
            end

            getgenv().EnableRemote = function(eventOrFunc)
                disabled[eventOrFunc] = false
            end

            getgenv().DisableNebulaRemoteAbuseAPI = function()
                disabled = {}
                returns = {}
                intercepted = {}
                mt.__namecall = old_namecall
            end

            mt.__namecall = newcclosure(function(self, ...)
                if InfStamEnabled == false then
                    return old_namecall(self, ...)
                end

                local args = {...}
                local event = namecallMethod()
                if methods[self.ClassName] == event then
                    if disabled[self] then
                        return
                    elseif intercepted[self] then
                        local intercept = intercepted[self]
                        if typeof(intercept) == "function" then
                            args = {intercept(unpack(args))}
                        elseif typeof(intercept) == "table" then
                            args = intercept
                        end
                    end
                end
                local returned = {old_namecall(self, unpack(args))}
                if returns[self] and returned then
                    returned = {returns[self](unpack(returned))}
                end
                return unpack(returned)
            end)

            local old_index = mt.__index
            mt.__index = function(o, k)
                if tostring(o) == "Humanoid" and tostring(k) == "WalkSpeed" then
                    return 16
                end
                return old_index(o, k)
            end

            local character = player.Character or player.CharacterAdded:Wait()
            local humanoid = character:WaitForChild("Humanoid")

            if ReplicatedStorage:FindFirstChild("FE") and ReplicatedStorage.FE:FindFirstChild("Sprint") then

                    return "Ended"
                end)
            end

                if InfStamEnabled == false then return end
                if key:lower() == "r" then
                    humanoid.WalkSpeed = 16

            end)

local function ccGetModule(tool, which)
    if not tool then return nil end
    local m = tool:FindFirstChild(which)
    if m then return m end
    for _, c in ipairs(tool:GetChildren()) do
        if c.Name:lower() == which:lower() then return c end
    end
    return nil
end
local function ccKeep1()
    local mod = ccGetModule(ccGkTool(), "Keep1Module")
    if not mod then return false end
    task.spawn(function() pcall(function() require(mod).Keep1() end) end)
    return true
end
local function ccSave()
    local mod = ccGetModule(ccGkTool(), "SaveModule")
    if not mod then return false end
    task.spawn(function() pcall(function() require(mod).Save() end) end)
    return true
end
local function ccJump()
    local RS = game:GetService("ReplicatedStorage")
    local mgr = RS:FindFirstChild("Managers")
    local bm = mgr and mgr:FindFirstChild("BManager")
    local ani = bm and bm:FindFirstChild("Animate")
    local jmp = ani and ani:FindFirstChild("jump")
    if jmp then
        local ok = pcall(function()
            local mod = require(jmp)
            if type(mod) == "function" then mod()
            elseif type(mod) == "table" and type(mod.Play) == "function" then mod.Play()
            elseif type(mod) == "table" and type(mod.Jump) == "function" then mod.Jump() end
        end)
        if ok then return true end
    end
    local char = player.Character
    local h = char and char:FindFirstChildOfClass("Humanoid")
    if h and h.Parent and h.Health > 0 then
        pcall(function()
            h:SetStateEnabled(Enum.HumanoidStateType.Jumping, true)
            h:SetStateEnabled(Enum.HumanoidStateType.Freefall, true)
            h.Jump = true
            h:ChangeState(Enum.HumanoidStateType.Jumping)
        end)
        return true
    end
    return false
end

local function ccRunCombo(id)

local function ccAddButton(def)
    local pos = CC.positions[def.id] or CC_DEFAULT_POS[def.id] or UDim2.new(0.5, 0, 0.5, 0)
    CC.positions[def.id] = pos
    local btn = Instance.new("TextButton")
    btn.Name = def.id
    btn.Size = UDim2.new(0, CC.btn.size, 0, CC.btn.size)
    btn.Position = pos
    btn.BackgroundColor3 = Color3.fromRGB(22, 12, 20)
    btn.BackgroundTransparency = CC.btn.transparency
    btn.TextColor3 = Color3.fromRGB(255, 105, 180)
    btn.TextTransparency = CC.btn.transparency
    btn.Font = Enum.Font.GothamBold
    btn.TextSize = 14
    btn.Text = def.label or def.id
    btn.BorderSizePixel = 0
    btn.Active = true
    btn.Visible = true
    btn.AutoButtonColor = false
    btn.Parent = ccGui
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 10)
    corner.Parent = btn
    local stroke = Instance.new("UIStroke")
    stroke.Color = CC.btn.locked and Color3.fromRGB(255, 105, 180) or Color3.fromRGB(255, 200, 60)
    stroke.Thickness = 1.4
    stroke.Transparency = math.clamp(0.35 + CC.btn.transparency, 0, 1)
    stroke.Parent = btn
    local dragging, dragStart, startPos, fired = false, nil, nil, false
    btn.InputBegan:Connect(function(input)
        if input.UserInputType ~= Enum.UserInputType.MouseButton1
        and input.UserInputType ~= Enum.UserInputType.Touch then return end
        if input.UserInputState and input.UserInputState ~= Enum.UserInputState.Begin then return end
        if CC.btn.locked then
            if CC.mobEnabled then ccRunCombo(def.combo) end
            fired = true
            return
        end
        dragging = true
        dragStart = input.Position
        startPos = btn.Position
        fired = false
    end)
    btn.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
            local d = input.Position - dragStart
            btn.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
    btn.InputEnded:Connect(function(input)
        if input.UserInputType ~= Enum.UserInputType.MouseButton1
        and input.UserInputType ~= Enum.UserInputType.Touch then return end
        if dragging then
            dragging = false
            CC.positions[def.id] = btn.Position
            if (input.Position - (dragStart or input.Position)).Magnitude < 6
            and not fired and CC.mobEnabled then
                ccRunCombo(def.combo)
            end
        end
        fired = false
    end)
    ccRefs[def.id] = { btn = btn, stroke = stroke }
end
for _, def in ipairs(CC_BTNS) do ccAddButton(def) end

local function ccSyncSizes()
    for _, r in pairs(ccRefs) do r.btn.Size = UDim2.new(0, CC.btn.size, 0, CC.btn.size) end
end
local function ccSyncOpacity()
    for _, r in pairs(ccRefs) do
        r.btn.BackgroundTransparency = CC.btn.transparency
        r.btn.TextTransparency = CC.btn.transparency
        if r.stroke then r.stroke.Transparency = math.clamp(0.35 + CC.btn.transparency, 0, 1) end
    end
end

    for _, r in pairs(ccRefs) do
        r.stroke.Color = CC.btn.locked and Color3.fromRGB(255, 105, 180) or Color3.fromRGB(255, 200, 60)
    end
end

end
