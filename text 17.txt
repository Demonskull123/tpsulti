local function ensureDir()
    if makefolder and isfolder and not isfolder(CONFIG_DIR) then makefolder(CONFIG_DIR) end
end

local function serialize(v)
    local t = typeof(v)
    if t == "Vector3" then return string.format("V3(%f,%f,%f)", v.X, v.Y, v.Z)
    elseif t == "Color3" then return string.format("C3(%d,%d,%d)", v.R * 255, v.G * 255, v.B * 255)
    elseif t == "UDim2" then return string.format("U2(%f,%d,%f,%d)", v.X.Scale, v.X.Offset, v.Y.Scale, v.Y.Offset)
    elseif t == "string" then return string.format("%q", v)
    elseif t == "boolean" then return tostring(v)
    elseif t == "number" then return tostring(v)
    elseif t == "nil" then return "nil" end
    return "nil"
end

local function deserialize(s)
    if s == "nil" or s == nil then return nil end
    if s == "true" then return true end
    if s == "false" then return false end
    local v3x, v3y, v3z = s:match("^V3%(([%d%.%-]+),([%d%.%-]+),([%d%.%-]+)%)$")
    if v3x then return Vector3.new(tonumber(v3x), tonumber(v3y), tonumber(v3z)) end
    local c3r, c3g, c3b = s:match("^C3%((%d+),(%d+),(%d+)%)$")
    if c3r then return Color3.fromRGB(tonumber(c3r), tonumber(c3g), tonumber(c3b)) end
    local u2xs, u2xo, u2ys, u2yo = s:match("^U2%(([%d%.%-]+),([%d%-]+),([%d%.%-]+),([%d%-]+)%)$")
    if u2xs then return UDim2.new(tonumber(u2xs), tonumber(u2xo), tonumber(u2ys), tonumber(u2yo)) end
    local num = tonumber(s)
    if num then return num end
    local str = s:match('^"(.*)"$')
    if str then return str end
    return s
end

local function saveConfig()
    ensureDir()
    local lines = {}
    for k, v in pairs(Config) do table.insert(lines, string.format('%s=%s', k, serialize(v))) end
    local data = table.concat(lines, "\n")
    if writefile then
        pcall(function() writefile(CONFIG_DIR .. "/" .. CONFIG_FILE, data) end)
        return true
    end
    return false
end

local function loadConfig()
    ensureDir()
    if not (isfile and readfile) then return false end
    if not isfile(CONFIG_DIR .. "/" .. CONFIG_FILE) then return false end
    local ok, data = pcall(function() return readfile(CONFIG_DIR .. "/" .. CONFIG_FILE) end)
    if not ok or not data or data == "" then return false end
    for line in data:gmatch("[^\n]+") do
        local k, v = line:match("^([^=]+)=(.*)$")
        if k and v then
            local val = deserialize(v)
            Config[k] = val
            if ConfigCallbacks[k] then pcall(ConfigCallbacks[k], val) end
        end
    end
    return true
end

local function register(key, ctl, default, apply)
    Config[key] = Config[key] ~= nil and Config[key] or default
    ConfigCallbacks[key] = function(v)
        Config[key] = v
        if apply then pcall(apply, v) end
        pcall(function() ctl:Set(v) end)
    end
end

local function SafeClear(tbl)
    if not tbl then return end
    if table.clear then table.clear(tbl) else for k in pairs(tbl) do tbl[k] = nil end end
end

local function GetReachDistanceForPart(partName)
    if partName == "Head" then
        if not reachSettings.HeadEnabled then return 0 end
        return reachSettings.ReachDistance * ((reachSettings.ReachX_Head + reachSettings.ReachY_Head + reachSettings.ReachZ_Head) / 3)
    elseif partName == "Right Arm" or partName == "Left Arm" then
        local side = partName:match("(%w+)%s")
        if not reachSettings[side .. "ArmEnabled"] then return 0 end
        return reachSettings.ReachDistance * ((reachSettings.ReachX_Arm + reachSettings.ReachY_Arm + reachSettings.ReachZ_Arm) / 3)
    elseif partName == "Right Leg" or partName == "Left Leg" then
        local side = partName:match("(%w+)%s")
        if not reachSettings[side .. "LegEnabled"] then return 0 end
        return reachSettings.ReachDistance * ((reachSettings.ReachX_Leg + reachSettings.ReachY_Leg + reachSettings.ReachZ_Leg) / 3)
    end
    return 0
end

local function CreateLimbHitbox(bodyPart, partName)
    local hb = Instance.new("Part")
    hb.Name = partName .. "_Hitbox"
    hb.Anchored = false
    hb.CanCollide = false
    hb.Transparency = 1
    hb.Massless = true
    if partName == "Head" then
        hb.Size = Vector3.new(bodyPart.Size.X * reachSettings.ReachX_Head * 2, bodyPart.Size.Y * reachSettings.ReachY_Head * 2, bodyPart.Size.Z * reachSettings.ReachZ_Head * 2)
    elseif partName == "Right Arm" or partName == "Left Arm" then
        hb.Size = Vector3.new(bodyPart.Size.X * reachSettings.ReachX_Arm * 2, bodyPart.Size.Y * reachSettings.ReachY_Arm * 2, bodyPart.Size.Z * reachSettings.ReachZ_Arm * 2)
    else
        hb.Size = Vector3.new(bodyPart.Size.X * reachSettings.ReachX_Leg * 2, bodyPart.Size.Y * reachSettings.ReachY_Leg * 2, bodyPart.Size.Z * reachSettings.ReachZ_Leg * 2)
    end
    hb.CFrame = bodyPart.CFrame
    hb.Parent = bodyPart
    local w = Instance.new("WeldConstraint")
    w.Part0 = bodyPart
    w.Part1 = hb
    w.Parent = hb
    return hb
end

local function GetVisTransparencyFor(partName)
    if partName == "Head" then return reachSettings.HeadVisTransparency
    elseif partName == "Right Arm" or partName == "Left Arm" then return reachSettings.ArmVisTransparency
    else return reachSettings.LegVisTransparency end
end

local function UpdateLimbVisualizer(partName)
    local char = player.Character
    if not char then return end
    local bodyPart = char:FindFirstChild(partName)
    if not bodyPart then return end
    local reachDist = GetReachDistanceForPart(partName)
    if reachDist > 0 then
        if LimbHitboxes[partName] then LimbHitboxes[partName]:Destroy() end
        LimbHitboxes[partName] = CreateLimbHitbox(bodyPart, partName)
    else
        if LimbHitboxes[partName] then LimbHitboxes[partName]:Destroy() LimbHitboxes[partName] = nil end
    end
    local visOn = false
    if partName == "Head" then visOn = reachSettings.HeadVisualizer
    elseif partName == "Right Arm" then visOn = reachSettings.RightArmVisualizer
    elseif partName == "Left Arm" then visOn = reachSettings.LeftArmVisualizer
    elseif partName == "Right Leg" then visOn = reachSettings.RightLegVisualizer
    elseif partName == "Left Leg" then visOn = reachSettings.LeftLegVisualizer end
    if LimbVisualizers[partName] then LimbVisualizers[partName]:Destroy() LimbVisualizers[partName] = nil end
    if visOn and LimbHitboxes[partName] then
        local hb = LimbHitboxes[partName]
        local vis = Instance.new("Part")
        vis.Name = partName .. "_Visualizer"
        vis.Size = hb.Size
        vis.CFrame = hb.CFrame
        vis.Anchored = false
        vis.CanCollide = false
        vis.Transparency = GetVisTransparencyFor(partName)
        vis.Color = Color3.fromRGB(0, 255, 100)
        vis.Material = Enum.Material.Neon
        vis.Massless = true
        vis.Parent = hb
        local w = Instance.new("WeldConstraint")
        w.Part0 = hb
        w.Part1 = vis
        w.Parent = vis
        LimbVisualizers[partName] = vis
    end
end

local function RefreshVisTransparency()
    for partName, v in pairs(LimbVisualizers) do
        if v and v.Parent then v.Transparency = GetVisTransparencyFor(partName) end
    end
end

local function FindBallPartLimb(part)
    if part:IsA("BasePart") and (part.Name == "TPS" or part.Name == "PSoccerBall" or BALL_SET[part.Name]) then return part end
    local cur = part
    while cur and cur ~= Workspace do
        if cur:IsA("BasePart") and (cur.Name == "PSoccerBall" or cur.Name == "TPS" or BALL_SET[cur.Name]) then return cur end
        cur = cur.Parent
    end
    return nil
end

    local char = player.Character
    if not char then return end
    for _, partName in ipairs(LimbBodyParts) do
        local bodyPart = char:FindFirstChild(partName)
        if bodyPart and GetReachDistanceForPart(partName) > 0 then
            if LimbHitboxes[partName] then LimbHitboxes[partName]:Destroy() end
            LimbHitboxes[partName] = CreateLimbHitbox(bodyPart, partName)
            UpdateLimbVisualizer(partName)
        end
    end
end

                        local key = partName .. ":" .. ballPart:GetFullName()
                        newTouches[key] = { part = ballPart, bodyPart = bodyPart, time = now }
                        if not LimbActiveTouches[key] or (now - LimbActiveTouches[key].time) > LIMB_TOUCH_COOLDOWN then
                            LimbActiveTouches[key] = { part = ballPart, bodyPart = bodyPart, time = now }
                            pcall(function()
                                firetouchinterest(ballPart, bodyPart, 0)
                                task.wait()
                                firetouchinterest(ballPart, bodyPart, 1)
                            end)
                        end
                    end
                end
            end
        end
    end
    for key, d in pairs(LimbActiveTouches) do
        if not newTouches[key] and (now - d.time) > LIMB_TOUCH_COOLDOWN then LimbActiveTouches[key] = nil end
    end
end

        if limb then
            AutoCatch_TouchLimb(limb, ball, AUTO_CATCH_ARM_FIRES)
            if (limb.Position - ball.Position).Magnitude <= (limb.Size.Magnitude * 0.75 + 1) then armHit = true end
        end
    end
    if armHit then
        AutoCatch_LastArmContact.time = os.clock()
        AutoCatch_LastArmContact.ball = ball
        return true
    end
    for _, name in ipairs({ "Right Leg", "Left Leg", "Head" }) do
        local limb = char:FindFirstChild(name)
        if limb then AutoCatch_TouchLimb(limb, ball, AUTO_CATCH_BODY_FIRES) end
    end
    return false
end

local function AutoCatch_Tick()
    if not AutoCatch.Enabled then return end
    local char = player.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local balls = {}
    for _, b in ipairs(TrackedBalls) do if b and b.Parent then table.insert(balls, b) end end
    local main = GetMainBall()
    if main and not table.find(balls, main) then table.insert(balls, main) end
    for _, b in ipairs(GetPracticeBalls()) do if not table.find(balls, b) then table.insert(balls, b) end end
    if #balls == 0 then return end
    for _, ball in ipairs(balls) do
        local dist = (ball.Position - root.Position).Magnitude
        if dist <= AutoCatch.Distance then
            AutoCatch_ArmsFirst(ball, char)
            for partName, hb in pairs(LimbHitboxes) do
                if hb and hb.Parent and (partName == "Right Arm" or partName == "Left Arm") then
                    local bodyPart = char:FindFirstChild(partName)
                    if bodyPart then AutoCatch_TouchLimb(bodyPart, ball, 3) end
                end
            end
            if AutoCatch.Force then
                local toRoot = (root.Position - ball.Position).Unit
                local cur = ball.AssemblyLinearVelocity
                local away = cur:Dot(-toRoot)
                if away > 0 then
                    local pull = math.min(away * (AutoCatch.PullMult or 0.85), AutoCatch.Cap or 55)
                    ball.AssemblyLinearVelocity = cur + toRoot * pull
                end
            end
            if AutoCatch.Dampen then pcall(function() ball.AssemblyAngularVelocity = ball.AssemblyAngularVelocity * 0.5 end) end
        end
    end
end

local function ccJump()
    local RS = game:GetService("ReplicatedStorage")
    local mgr = RS:FindFirstChild("Managers")
    local bm = mgr and mgr:FindFirstChild("BManager")
    local ani = bm and bm:FindFirstChild("Animate")
    local jmp = ani and ani:FindFirstChild("jump")
    if jmp then
        local ok = pcall(function()
            local mod = require(jmp)
            if type(mod) == "function" then mod()
            elseif type(mod) == "table" and type(mod.Play) == "function" then mod.Play()
            elseif type(mod) == "table" and type(mod.Jump) == "function" then mod.Jump() end
        end)
        if ok then return true end
    end
    local char = player.Character
    local h = char and char:FindFirstChildOfClass("Humanoid")
    if h and h.Parent and h.Health > 0 then
        pcall(function()
            h:SetStateEnabled(Enum.HumanoidStateType.Jumping, true)
            h:SetStateEnabled(Enum.HumanoidStateType.Freefall, true)
            h.Jump = true
            h:ChangeState(Enum.HumanoidStateType.Jumping)
        end)
        return true
    end
    return false
end

    btn.InputBegan:Connect(function(input)
        if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then return end
        if input.UserInputState and input.UserInputState ~= Enum.UserInputState.Begin then return end
        if CC.btn.locked then
            if CC.mobEnabled then ccRunCombo(def.combo) end
            fired = true
            return
        end
        dragging = true
        dragStart = input.Position
        startPos = btn.Position
        fired = false
    end)
    btn.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            local d = input.Position - dragStart
            btn.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)

        end
        fired = false
