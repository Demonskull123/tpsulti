-- REACH
local reachSettings = {
    HeadEnabled=false, HeadVisualizer=false, ReachX_Head=1, ReachY_Head=1, ReachZ_Head=1,
    RightArmEnabled=false, RightArmVisualizer=false, ReachX_Arm=1, ReachY_Arm=1, ReachZ_Arm=1,
    LeftArmEnabled=false, LeftArmVisualizer=false,
    RightLegEnabled=false, RightLegVisualizer=false, ReachX_Leg=1, ReachY_Leg=1, ReachZ_Leg=1,
    LeftLegEnabled=false, LeftLegVisualizer=false,
    ReachDistance=50,
}
local BodyParts = {"Head","Right Arm","Left Arm","Right Leg","Left Leg"}
local Character = player.Character or player.CharacterAdded:Wait()
local Hitboxes, Visualizers, ActiveTouches = {}, {}, {}
local TOUCH_COOLDOWN = 0.1

local function GetReachDistanceForPart(p)
    if p == "Head" then
        if not reachSettings.HeadEnabled then return 0 end
        return reachSettings.ReachDistance * (reachSettings.ReachX_Head + reachSettings.ReachY_Head + reachSettings.ReachZ_Head) / 3
    elseif p == "Right Arm" or p == "Left Arm" then
        local side = p:match("(%w+)%s")
        if not reachSettings[side.."ArmEnabled"] then return 0 end
        return reachSettings.ReachDistance * (reachSettings.ReachX_Arm + reachSettings.ReachY_Arm + reachSettings.ReachZ_Arm) / 3
    elseif p == "Right Leg" or p == "Left Leg" then
        local side = p:match("(%w+)%s")
        if not reachSettings[side.."LegEnabled"] then return 0 end
        return reachSettings.ReachDistance * (reachSettings.ReachX_Leg + reachSettings.ReachY_Leg + reachSettings.ReachZ_Leg) / 3
    end
    return 0
end

local function CreateHitbox(bp, name)
    local hb = Instance.new("Part")
    hb.Name = name.."_Hitbox" hb.Anchored = false hb.CanCollide = false hb.Transparency = 1 hb.Massless = true
    if name == "Head" then
        hb.Size = Vector3.new(bp.Size.X * reachSettings.ReachX_Head * 2, bp.Size.Y * reachSettings.ReachY_Head * 2, bp.Size.Z * reachSettings.ReachZ_Head * 2)
    elseif name == "Right Arm" or name == "Left Arm" then
        hb.Size = Vector3.new(bp.Size.X * reachSettings.ReachX_Arm * 2, bp.Size.Y * reachSettings.ReachY_Arm * 2, bp.Size.Z * reachSettings.ReachZ_Arm * 2)
    else
        hb.Size = Vector3.new(bp.Size.X * reachSettings.ReachX_Leg * 2, bp.Size.Y * reachSettings.ReachY_Leg * 2, bp.Size.Z * reachSettings.ReachZ_Leg * 2)
    end
    hb.CFrame = bp.CFrame hb.Parent = bp
    local w = Instance.new("WeldConstraint") w.Part0 = bp w.Part1 = hb w.Parent = hb
    return hb
end

function UpdateVisualizer(name)
    local bp = Character:FindFirstChild(name)
    if not bp then return end
    if GetReachDistanceForPart(name) > 0 then
        if Hitboxes[name] then Hitboxes[name]:Destroy() end
        Hitboxes[name] = CreateHitbox(bp, name)
    else
        if Hitboxes[name] then Hitboxes[name]:Destroy() Hitboxes[name] = nil end
    end
    local vis = (name == "Head" and reachSettings.HeadVisualizer) or (name == "Right Arm" and reachSettings.RightArmVisualizer) or (name == "Left Arm" and reachSettings.LeftArmVisualizer) or (name == "Right Leg" and reachSettings.RightLegVisualizer) or (name == "Left Leg" and reachSettings.LeftLegVisualizer)
    if Visualizers[name] then Visualizers[name]:Destroy() Visualizers[name] = nil end
    if vis and Hitboxes[name] then
        local v = Instance.new("Part")
        v.Name = name.."_Visualizer" v.Size = Hitboxes[name].Size v.CFrame = Hitboxes[name].CFrame
        v.Anchored = false v.CanCollide = false v.Transparency = 0.7 v.Color = Color3.fromRGB(0,255,0) v.Material = Enum.Material.Neon v.Massless = true
        v.Parent = Hitboxes[name]
        local w = Instance.new("WeldConstraint") w.Part0 = Hitboxes[name] w.Part1 = v w.Parent = v
        Visualizers[name] = v
    end
end

local function FindBallPart(part)
    if part:IsA("BasePart") and (part.Name == "TPS" or part.Name == "PSoccerBall") then return part end
    local cur = part
    while cur and cur ~= workspace do
        if cur:IsA("BasePart") and cur.Name == "PSoccerBall" then return cur end
        cur = cur.Parent
    end
    return nil
end

local function CreateAllHitboxes()
    for _, n in ipairs(BodyParts) do
        local bp = Character:FindFirstChild(n)
        if bp and GetReachDistanceForPart(n) > 0 then
            if Hitboxes[n] then Hitboxes[n]:Destroy() end
            Hitboxes[n] = CreateHitbox(bp, n)
        end
    end
end

local function RemoveAllHitboxes()
    for _, hb in pairs(Hitboxes) do if hb then hb:Destroy() end end
    for _, v in pairs(Visualizers) do if v then v:Destroy() end end
    table.clear(Hitboxes) table.clear(Visualizers) table.clear(ActiveTouches)
end

local function CheckBallCollisions()
    if not Character or not Character.Parent then return end
    local now = tick() local newTouches = {}
    for name, hb in pairs(Hitboxes) do
        if hb and hb.Parent then
            local body = Character:FindFirstChild(name)
            if body then
                local op = OverlapParams.new() op.FilterDescendantsInstances = {Character} op.FilterType = Enum.RaycastFilterType.Exclude
                for _, p in ipairs(workspace:GetPartBoundsInBox(hb.CFrame, hb.Size, op)) do
                    local ball = FindBallPart(p)
                    if ball then
                        local key = name..":"..ball:GetFullName()
                        newTouches[key] = true
                        if not ActiveTouches[key] or (now - ActiveTouches[key]) > TOUCH_COOLDOWN then
                            ActiveTouches[key] = now
                            pcall(function() firetouchinterest(ball, body, 0) firetouchinterest(ball, body, 1) end)
                        end
                    end
                end
            end
        end
    end
    for k, t in pairs(ActiveTouches) do if not newTouches[k] and (now - t) > TOUCH_COOLDOWN then ActiveTouches[k] = nil end end
end

    local now = tick()
    if now - lastUpdate >= 0.1 then lastUpdate = now pcall(CheckBallCollisions) end
end)

                    local vel = ball.AssemblyLinearVelocity
                    if vel.Magnitude >= minBallSpeed then
                        local pp = ball.Position + vel * pt
                        local pd = (pp - hrp.Position).Magnitude
                        local cd = (ball.Position - hrp.Position).Magnitude
                        local d = math.min(cd, pd)
                        local dir = (ball.Position - hrp.Position).Unit
                        local bdir = vel.Unit
                        local ang = dir:Dot(bdir)
                        if d < bd and (ang > -0.1 or pd < 12) then bd = d best = ball end
                    end
                end
            end

local function ccApplyPayload(d)
    if type(d) ~= "table" then return end
    if type(d.btn) == "table" then
        if type(d.btn.size) == "number" then CC.btn.size = d.btn.size end
        if type(d.btn.opacity) == "number" then CC.btn.opacity = d.btn.opacity end
        if type(d.btn.locked) == "boolean" then CC.btn.locked = d.btn.locked end
    end
    if type(d.positions) == "table" then
        for id, p in pairs(d.positions) do
            if type(p) == "table" and p.xs and p.xo and p.ys and p.yo then CC.positions[id] = UDim2.new(p.xs, p.xo, p.ys, p.yo) end
        end
    end
    if type(d.timings) == "table" then
        for k, v in pairs(d.timings) do if CC[k] ~= nil and type(v) == type(CC[k]) then CC[k] = v end end
    end
end

local function ccProbe(url)
    local ok, res = pcall(function() return game:HttpGet(url) end)
    if not ok then return false, "HttpGet failed: "..tostring(res) end
    if type(res) ~= "string" then return false, "HttpGet returned "..typeof(res) end
    if #res == 0 then return false, "empty body" end
    local head = res:sub(1,200):lower()
    if head:find("<!doctype") or head:find("<html") or head:find("<head") then return false, "URL returned HTML not Lua:\n\n"..res:sub(1,200) end
    return true, res
end
local function ccLoadCustom()
    if customLoaded then return true end
    local ok, body = ccProbe(customURL)
    if not ok then warn("[TPSULT] "..body) ccShowPopup("Custom Controls: fetch failed", body) return false end
    local ls = loadstring or load
    if type(ls) ~= "function" then ccShowPopup("Custom Controls: execute failed", "no loadstring/load") return false end
    local fn, perr = ls(body)
    if not fn then
        local msg = "compile error from "..customURL..":\n\n"..tostring(perr)
        ccShowPopup("Custom Controls: execute failed", msg) return false
    end
    local ran, rerr = pcall(fn)
    if not ran then
        local msg = "runtime error from "..customURL..":\n\n"..tostring(rerr)
        ccShowPopup("Custom Controls: execute failed", msg) return false
    end
    customLoaded = true
    return true
end
