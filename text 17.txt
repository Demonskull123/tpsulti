pcall(function()
    local mt = getrawmetatable(game)
    if not mt then return end
    setreadonly(mt, false)
    local oldNC = mt.__namecall
    mt.__namecall = newcclosure(function(self, ...)
        if not AntiBan.Enabled then return oldNC(self, ...) end
        local method = getnamecallmethod()
        if method == "Kick" and self == LocalPlayer then return nil end
        if method == "FireServer" or method == "InvokeServer" then
            local ok, name = pcall(function() return self.Name end)
            if ok and type(name) == "string" then
                local n = name:lower()
                if n:find("anticheat") or n:find("detect") or n:find("ban")
                   or n:find("flag") or n:find("report") or n:find("vote")
                   or n:find("kick") then
                    return nil
                end
            end
        end
        return oldNC(self, ...)
    end)
    setreadonly(mt, true)
end)

-- Destruction anti-cheat scripts
pcall(function()
    local Destroyed = setmetatable({}, {__mode = "k"})
    local Keywords = {"Banned", "Walkspeed", "Ban", "AntiCheat"}
    local function ContainsBad(text)
        for _, k in ipairs(Keywords) do
            if text:find(k, 1, true) then return true end
        end
        return false
    end
    local function IsAntiCheat(scr)
        if typeof(scr) ~= "Instance" or not scr:IsA("LuaSourceContainer") then return false end
        local ok, src = pcall(function() return scr.Source end)
        return ok and typeof(src) == "string" and ContainsBad(src)
    end
    for _, d in ipairs(game:GetDescendants()) do
        if (d:IsA("LocalScript") or d:IsA("ModuleScript") or
            (d:IsA("Script") and d.RunContext == Enum.RunContext.Client)) and
            not Destroyed[d] and IsAntiCheat(d) then
            pcall(function() d:Destroy() end)
            Destroyed[d] = true
        end
    end
    game.DescendantAdded:Connect(function(d)
        if (d:IsA("LocalScript") or d:IsA("ModuleScript") or d:IsA("Script")) and IsAntiCheat(d) then
            pcall(function() d:Destroy() end)
            Destroyed[d] = true
        end
    end)
end)

local function FindAllBalls()
    TrackedBalls = {}
    local tps = Workspace:FindFirstChild("TPSSystem")
    if tps then
        local tpsBall = tps:FindFirstChild("TPS")
        if tpsBall and tpsBall:IsA("BasePart") then table.insert(TrackedBalls, tpsBall) end
    end
    local mps = Workspace:FindFirstChild("MPSSystem") or Workspace:FindFirstChild("MPS")
    if mps then
        for _, child in ipairs(mps:GetChildren()) do
            if child:IsA("BasePart") and (child.Name == "MPSBall" or child.Name == "Ball" or child.Name == "MPS") then
                table.insert(TrackedBalls, child)
            end
        end
    end
    local practice = Workspace:FindFirstChild("Practice")
    if practice then
        for _, child in ipairs(practice:GetChildren()) do
            if child:IsA("BasePart") and (child.Name == "PSoccerBall" or child.Name == "Ball") then
                table.insert(TrackedBalls, child)
            end
        end
    end
    for _, n in ipairs(BALL_NAMES) do
        local b = Workspace:FindFirstChild(n)
        if b and b:IsA("BasePart") and not table.find(TrackedBalls, b) then
            table.insert(TrackedBalls, b)
        end
    end
end

local function GetMainBall()
    local now = os.clock()
    if Cache.MainBall and Cache.MainBall.Parent and (now - Cache.LastBallScan) < 0.5 then
        return Cache.MainBall
    end
    local tps = Workspace:FindFirstChild("TPSSystem")
    if tps then
        local b = tps:FindFirstChild("TPS")
        if b then Cache.MainBall = b Cache.LastBallScan = now return b end
    end
    for _, name in ipairs(BALL_NAMES) do
        local ball = Workspace:FindFirstChild(name)
        if ball and ball:IsA("BasePart") then
            Cache.MainBall = ball Cache.LastBallScan = now return ball
        end
    end
    Cache.MainBall = nil
    return nil
end

local function GetPracticeBalls()
    local list = {}
    local practice = Workspace:FindFirstChild("Practice")
    if practice then
        for _, child in ipairs(practice:GetChildren()) do
            if child:IsA("BasePart") and (child.Name == "PSoccerBall" or child.Name == "Ball") then
                table.insert(list, child)
            end
        end
    end
    return list
end

local function GetBallList()
    local now = os.clock()
    if (now - Cache.LastBallListScan) < 0.5 and #Cache.BallList > 0 then
        local valid = true
        for _, b in ipairs(Cache.BallList) do
            if not b.Parent then valid = false break end
        end
        if valid then return Cache.BallList end
    end
    local balls, seen = {}, {}
    local tps = Workspace:FindFirstChild("TPSSystem")
    if tps then
        local tpsBall = tps:FindFirstChild("TPS")
        if tpsBall and tpsBall:IsA("BasePart") then
            table.insert(balls, tpsBall) seen[tpsBall] = true
        end
    end
    local practice = Workspace:FindFirstChild("Practice")
    if practice then
        for _, child in ipairs(practice:GetChildren()) do
            if child:IsA("BasePart") and (child.Name == "PSoccerBall" or child.Name == "Ball") then
                if not seen[child] then table.insert(balls, child) seen[child] = true end
            end
        end
    end
    for _, n in ipairs(BALL_NAMES) do
        local b = Workspace:FindFirstChild(n)
        if b and b:IsA("BasePart") and not seen[b] then
            table.insert(balls, b) seen[b] = true
        end
    end
    Cache.BallList = balls
    Cache.LastBallListScan = now
    return balls
end

    Ultra  = { ping = 2,  jitter = 1  },

local NetworkOK = false
local NetworkSettings = nil
pcall(function()
    NetworkSettings = settings():GetService("NetworkSettings")
    NetworkOK = NetworkSettings and NetworkSettings.IncomingReplicationLag ~= nil
end)

local function ApplyMode(mode)
    if not NetworkOK then Notify("NetworkSettings non supporte", 3) return end
    local p = MODE_PRESETS[mode] or MODE_PRESETS.Normal
    pcall(function()
        NetworkSettings.IncomingReplicationLag = p.ping / 1000
        pcall(function() NetworkSettings.InboundNetworkMinDelayMs = p.ping end)
        pcall(function() NetworkSettings.InboundNetworkJitterMs = p.jitter end)
    end)
    UpdateLabels(mode, true)
    Notify("Mode: " .. mode .. " (" .. p.ping .. "ms)", 2)
end

local function ResetMode()
    if not NetworkOK then return end
    pcall(function()
        NetworkSettings.IncomingReplicationLag = 0
        pcall(function() NetworkSettings.InboundNetworkMinDelayMs = 0 end)
        pcall(function() NetworkSettings.InboundNetworkJitterMs = 0 end)
    end)
    UpdateLabels("Normal", false)
    Notify("Delay Reducer OFF", 2)
end

local function RemoveCustomSky()
    pcall(function()

local function ApplySkybox(name)
    if not SkyboxCfg.Enabled then return end
    pcall(function()
        local preset = SKYBOX_PRESETS[name]
        RemoveCustomSky()
        if preset == nil then
            if SkyboxCfg.OriginalSky then SkyboxCfg.OriginalSky.Parent = SkyboxCfg.OriginalSkyParent or Lighting end
            return
        end
        if SkyboxCfg.OriginalSky then SkyboxCfg.OriginalSky.Parent = nil end
        local sky = Instance.new("Sky")
        sky.Name = "FireHub_Sky"
        for k, v in pairs(preset) do pcall(function() sky[k] = v end) end
        sky.Parent = Lighting
    end)
end

local Esp = { Enabled = false, Color = Color3.fromRGB(255,105,180), Transparency = 0.3, Highlights = {} }
local Predict = {
    Enabled = false, Time = 0.15, Steps = 15,
    Color = Color3.fromRGB(255,105,180), Transparency = 0.5, Size = 0.5,
    UseGravity = true, Parts = {}, Active = 0, RainbowActive = false,
}

local function ClearEsp()
    for _, h in pairs(Esp.Highlights) do pcall(function() h:Destroy() end) end
    Esp.Highlights = {}
end

local function SyncEsp()
    if not Esp.Enabled then return end
    local balls = GetBallList()
    for ball, h in pairs(Esp.Highlights) do
        if not ball.Parent or not h.Parent then
            pcall(function() h:Destroy() end)
            Esp.Highlights[ball] = nil
        end
    end
    for _, ball in ipairs(balls) do
        local h = Esp.Highlights[ball]
        if not h then
            h = Instance.new("Highlight")
            h.Name = "FireHub_ESP"
            h.FillColor = Esp.Color
            h.FillTransparency = Esp.Transparency
            h.OutlineColor = Esp.Color
            h.OutlineTransparency = 0
            h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            h.Adornee = ball
            h.Parent = CoreGui
            Esp.Highlights[ball] = h
        end
    end
end

local function EnsurePredictPool(count)
    local current = #Predict.Parts
    if current >= count then return end
    for i = current + 1, count do
        local part = Instance.new("Part")
        part.Name = "FireHub_Predict"
        part.Shape = Enum.PartType.Ball
        part.Anchored = true
        part.CanCollide = false
        part.CanQuery = false
        part.CanTouch = false
        part.CastShadow = false
        part.Material = Enum.Material.Neon
        part.Parent = Workspace
        Predict.Parts[i] = part
    end
end

local function SyncPrediction()
    if not Predict.Enabled then
        if Predict.Active > 0 then
            for i = 1, #Predict.Parts do
                if Predict.Parts[i] then Predict.Parts[i].Transparency = 1 end
            end
            Predict.Active = 0
        end
        return
    end
    local balls = GetBallList()
    local steps = Predict.Steps
    local needed = #balls * steps
    if needed == 0 then return end
    EnsurePredictPool(needed)
    local gravityY = -Workspace.Gravity
    local useG = Predict.UseGravity
    local sizeVec = Vector3.new(Predict.Size, Predict.Size, Predict.Size)
    local idx = 0
    for _, ball in ipairs(balls) do
        if ball.Parent then
            local pos, vel = ball.Position, ball.AssemblyLinearVelocity
            for i = 1, steps do
                idx = idx + 1
                local p = Predict.Parts[idx]
                if not p then break end
                local t = (Predict.Time / steps) * i
                local px, py, pz
                if useG then
                    local half_t2 = 0.5 * (t * t)
                    px = pos.X + vel.X * t
                    py = pos.Y + vel.Y * t + gravityY * half_t2
                    pz = pos.Z + vel.Z * t
                else
                    px = pos.X + vel.X * t
                    py = pos.Y + vel.Y * t
                    pz = pos.Z + vel.Z * t
                end
                p.Size = sizeVec
                p.Position = Vector3.new(px, py, pz)

local BallSizeCfg = { Size = 2.3, PracticeSize = 2.3, Conn = nil, PracticeConn = nil, OriginalSizes = {} }
local function ApplyBallSize()
    pcall(function()
        local ball = GetMainBall()
        if ball and ball:IsA("BasePart") then
            if not BallSizeCfg.OriginalSizes[ball] then BallSizeCfg.OriginalSizes[ball] = ball.Size end
            ball.Size = Vector3.new(BallSizeCfg.Size, BallSizeCfg.Size, BallSizeCfg.Size)
        end
    end)
end
local function ApplyPracticeBallSize()
    pcall(function()
        for _, b in ipairs(GetPracticeBalls()) do
            if not BallSizeCfg.OriginalSizes[b] then BallSizeCfg.OriginalSizes[b] = b.Size end
            b.Size = Vector3.new(BallSizeCfg.PracticeSize, BallSizeCfg.PracticeSize, BallSizeCfg.PracticeSize)
        end
    end)
end
local function RestoreOriginalSizes()
    pcall(function()
        for part, originalSize in pairs(BallSizeCfg.OriginalSizes) do
            if part and part.Parent then part.Size = originalSize end
        end
        BallSizeCfg.OriginalSizes = {}
    end)
end
