local FS = {
    available  = type(writefile) == "function" and type(readfile) == "function",
    baseFolder = "TempoHub",
    filePath   = "TempoHub/lastSword.txt",
}

local function ensureFolder()
    if not FS.available then return end
    pcall(function()
        if type(isfolder) == "function" and type(makefolder) == "function" then
            if not isfolder(FS.baseFolder) then
                makefolder(FS.baseFolder)
            end
        end
    end)
end

local function writeFS(path, data)
    if not FS.available then return false end
    ensureFolder()
    return pcall(function() writefile(path, tostring(data)) end)
end

local function readFS(path)
    if not FS.available then return nil end
    local ok, data = pcall(function()
        if type(isfile) == "function" then
            if not isfile(path) then return nil end
            return readfile(path)
        end
        return readfile(path)
    end)
    if ok and type(data) == "string" and data ~= "" then return data end
    return nil
end

-- FPS TRACKER
-- ============================================================
local FPS = {
    frames = 0, elapsed = 0, current = 60, smoothed = 60,
    instant = 60, lastDt = 1/60, spikeActive = false,
    spikeStartTime = 0, lowestRecent = 60, lowestRecentTime = 0,
}

local function updateFps(dt)
    local clamped = math.clamp(dt, 0.001, FPS_SMOOTH_DT)
    if clamped > 0.001 then
        FPS.instant = math.clamp(1 / clamped, 5, 500)
        FPS.lastDt  = clamped
    end
    local now = os.clock()
    if dt > FPS_SPIKE_DT then
        FPS.spikeActive = true
        FPS.spikeStartTime = now
        if FPS.instant < FPS.lowestRecent or now - FPS.lowestRecentTime > 0.5 then
            FPS.lowestRecent = FPS.instant
            FPS.lowestRecentTime = now
        end
    elseif FPS.spikeActive and now - FPS.spikeStartTime > 0.3 then
        FPS.spikeActive = false
    end
    FPS.frames  = FPS.frames + 1
    FPS.elapsed = FPS.elapsed + clamped
    if FPS.elapsed >= 0.25 then
        local raw = FPS.frames / FPS.elapsed
        FPS.smoothed = FPS.smoothed + (raw - FPS.smoothed) * 0.3
        FPS.current  = raw
        FPS.frames   = 0
        FPS.elapsed  = 0
    end
    if CFG.fpsOverride > 0 then return CFG.fpsOverride end
    return math.clamp(FPS.smoothed, 10, 360)
end

local function getFps()
    if CFG.fpsOverride > 0 then return CFG.fpsOverride end
    if not CFG.fastSpikeResponse then
        return math.clamp(FPS.smoothed, 10, 360)
    end
    local v = math.min(FPS.smoothed, FPS.instant)
    if FPS.spikeActive then v = math.min(v, FPS.lowestRecent) end
    return math.clamp(v, 5, 360)
end

local function fpsComp(fps)
    if not CFG.useFpsComp then return 0 end
    local ratio = FPS_TARGET / math.max(fps, 5)
    return math.min(ratio * CFG.fpsCompFactor, CFG.fpsCompMax)
end

local function autoFpsTier(fps)
    for _, t in ipairs(FPS_TIERS) do
        if fps >= t.minFps then
            CFG.fpsCompFactor = t.factor
            CFG.fpsCompMax   = t.cap
            return
        end
    end
end

local parryRemote = nil

local function findParryRemote()
    if parryRemote and parryRemote.Parent then return true end
    local remotes = ReplicatedStorage:FindFirstChild("Remotes")
    if not remotes then return false end
    local r = remotes:FindFirstChild("ParryButtonPress")
    if r then

        return true

local function parryMouse()
    pcall(function()
        VirtualInputManager:SendMouseButtonEvent(0, 0, 0, true, game, 0)
        task.wait(0.01)
        VirtualInputManager:SendMouseButtonEvent(0, 0, 0, false, game, 0)
    end)
end

local function parryRemoteFire()
    if not findParryRemote() then return end
    pcall(function() parryRemote:Fire() end)
end

local function parryKey()
    if not hasKeypress then
        parryMouse()
        return
    end
    pcall(function()
        keypress(0x46)
        task.wait(0.01)
        keyrelease(0x46)
    end)
end

local function doParry()
    local m = CFG.parryMethod
    if m == "Mouse" then
        parryMouse()
    elseif m == "Remote" then
        parryRemoteFire()
    elseif m == "Keypress" then
        parryKey()
    else
        parryRemoteFire()
        parryMouse()
    end
end

    ball = nil, pos = nil, speed = 0, smooth = 0,

local function resetBallTrack()
    ballTrack.ball       = nil
    ballTrack.pos        = nil
    ballTrack.speed      = 0
    ballTrack.smooth     = 0
    ballTrack.accel      = 0
    ballTrack.hasSample  = false
    ballTrack.lastImpact = 0
end

local function pushTargetHistory(target)
    if not target then return end
    local now = os.clock()
    for i = #targetHistory, 1, -1 do
        if now - targetHistory[i].time > CLASH_HISTORY then
            table.remove(targetHistory, i)
        end
    end
    local last = targetHistory[#targetHistory]
    if not last or last.target ~= target then
        table.insert(targetHistory, { time = now, target = target })
    end
end

local function nearbyPlayers(ball, radius)
    local set = {}
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            local char = plr.Character
            local root = char and char:FindFirstChild("HumanoidRootPart")
            if root and (root.Position - ball.Position).Magnitude <= radius then
                set[plr.Name] = true
            end
        end
    end
    return set
end

local function isClash(ball, threshold, radius)
    if not ball then return false end
    local near = nearbyPlayers(ball, radius)
    if next(near) == nil then return false end
    local count = 0
    for _, e in ipairs(targetHistory) do
        if e.target == LocalPlayer.Name or near[e.target] then
            count = count + 1
        end
    end
    return count >= threshold
end

local function updateClash(ball)
    if not CFG.autoClash or not ball then
        clashActive = false
        return
    end
    local hit = isClash(ball, CFG.clashExchangeThreshold, CFG.clashDetectRadius)
    local now = os.clock()
    if hit then
        clashTime   = now
        clashActive = true
    elseif now - clashTime > CLASH_FADE then
        clashActive = false
    end
end

local function getBallVelocity(ball, dt)
    local zoom = ball:FindFirstChild("zoomies")
    if zoom then
        local ok, vel = pcall(function() return zoom.VectorVelocity end)
        if ok and typeof(vel) == "Vector3" and vel.Magnitude > 0.1 then
            return vel
        end
    end
    local lin = ball.AssemblyLinearVelocity
    if lin.Magnitude > 0.1 then return lin end
    if ballTrack.pos and dt > 0 then
        return (ball.Position - ballTrack.pos) / dt
    end
    return Vector3.zero
end

local function timeToImpact(dist, speed, accel)
    if speed <= 0.01 then return math.huge end
    if accel > 0.5 then
        local disc = speed * speed + 2 * accel * dist
        return (-speed + math.sqrt(disc)) / accel
    end
    return dist / speed
end

local camCurve = {
    active = false, startTime = 0, duration = CURVE_DURATION,
    yaw = 0, pitch = 0, roll = 0, fovBoost = 0, originalFov = nil,
}

local function stopCamCurve()
    camCurve.active   = false
    camCurve.yaw      = 0
    camCurve.pitch    = 0
    camCurve.roll     = 0
    camCurve.fovBoost = 0
    if camCurve.originalFov then
        local cam = Workspace.CurrentCamera
        if cam then
            pcall(function() cam.FieldOfView = camCurve.originalFov end)
        end
    end
    camCurve.originalFov = nil
end

local function startCamCurve(mode)
    if mode == "Off" or not mode then return end
    local strength = CFG.curveDeflectStrength / 100
    if strength <= 0 then return end
    local dir = mode
    if mode == "Random" or mode == "Camera" then
        local opts = { "Right", "Left", "Up", "Down" }
        dir = opts[math.random(1, 4)]
    end
    local yawAmt   = CURVE_YAW   * strength
    local pitchAmt = CURVE_PITCH * strength
    local fovAmt   = CURVE_FOV   * strength
    local yaw, pitch, roll = 0, 0, 0
    if dir == "Right" then
        yaw  =  yawAmt
        roll = -pitchAmt
    elseif dir == "Left" then
        yaw  = -yawAmt
        roll =  pitchAmt
    elseif dir == "Up" then
        pitch = yawAmt
    elseif dir == "Down" then
        pitch = -yawAmt
    else
        return
    end
    camCurve.active    = true
    camCurve.startTime = os.clock()
    camCurve.duration  = CURVE_DURATION
    camCurve.yaw       = yaw
    camCurve.pitch     = pitch
    camCurve.roll      = roll
    camCurve.fovBoost  = fovAmt
    local cam = Workspace.CurrentCamera
    if cam and not camCurve.originalFov then
        camCurve.originalFov = cam.FieldOfView
    end
end

local function stepCamCurve()
    if not camCurve.active then return end
    local t = os.clock() - camCurve.startTime
    if t >= camCurve.duration then
        stopCamCurve()
        return
    end
    local frac = t / camCurve.duration
    local ease = (1 - frac) * (1 - frac)
    local cam  = Workspace.CurrentCamera
    if not cam then return end
    pcall(function()
        cam.CFrame = cam.CFrame * CFrame.Angles(
            math.rad(camCurve.pitch * ease),
            math.rad(camCurve.yaw   * ease),
            math.rad(camCurve.roll  * ease)
        )
    end)
    if camCurve.originalFov and camCurve.fovBoost ~= 0 then
        pcall(function()
            cam.FieldOfView = camCurve.originalFov + camCurve.fovBoost * ease
        end)
    end
end

pcall(function() RunService:UnbindFromRenderStep(CAMERA_BIND) end)
pcall(function()
    RunService:BindToRenderStep(CAMERA_BIND, Enum.RenderPriority.Camera.Value + 1, stepCamCurve)
end)

local ability = {
    active = false, type = nil, startTime = 0, duration = 0,
    windowMod = 1.0, lastDir = Vector3.zero, detectionCooldown = 0,
    avgSpeed = 0, preDeflectSpeed = 0, lastDeflector = nil,
    lastDeflectTime = 0, lastDeflectBall = nil, lastBallTarget = nil,
    lostTargetAt = 0, lastBall = nil,
}

local function getAbility(plr)
    if not plr then return nil end
    for _, attr in ipairs(ABILITY_ATTRS) do
        local ok, val = pcall(function() return plr:GetAttribute(attr) end)
        if ok and typeof(val) == "string" and val ~= "" then return val end
    end
    local char = plr.Character
    if char then
        for _, attr in ipairs(ABILITY_ATTRS) do
            local ok, val = pcall(function() return char:GetAttribute(attr) end)
            if ok and typeof(val) == "string" and val ~= "" then return val end
        end
    end
    return nil
end

local function isDangerAbility(name)
    if not name then return false end
    local lower = name:lower()
    for _, s in ipairs(ABILITY_SAFE) do
        if lower:find(s, 1, true) then return false end
    end
    for _, d in ipairs(ABILITY_DANGER) do
        if lower:find(d, 1, true) then return true end
    end
    return false
end

local function recordPreDeflect(ball)
    if not ball then return end
    local spd = ball.AssemblyLinearVelocity.Magnitude
    if spd > 0.1 then ability.preDeflectSpeed = spd end
end

local function detectSpike(ball)
    if not ball then return false, 0 end
    if ability.preDeflectSpeed <= 0.1 then return false, 0 end
    local now   = ball.AssemblyLinearVelocity.Magnitude
    local ratio = (now - ability.preDeflectSpeed) / ability.preDeflectSpeed
    return ratio >= CFG.abilitySpikeThreshold, ratio
end

local function onTargetChange(oldTarget, newTarget, ball)
    if oldTarget and oldTarget ~= LocalPlayer.Name and oldTarget ~= newTarget then
        if ball then recordPreDeflect(ball) end
        local plr = Players:FindFirstChild(oldTarget)
        if plr then
            local ab = getAbility(plr)
            if ab and isDangerAbility(ab) then
                ability.lastDeflector  = oldTarget
                ability.lastDeflectTime = os.clock()
                ability.lastDeflectBall = ball
            else
                ability.lastDeflector  = nil
                ability.lastDeflectTime = 0
            end
        end
    end
end

local function resetAbility()
    ability.active          = false
    ability.type            = nil
    ability.startTime       = 0
    ability.duration        = 0
    ability.windowMod       = 1.0
    ability.lastDir         = Vector3.zero
    ability.detectionCooldown = 0
    ability.avgSpeed        = 0
    ability.preDeflectSpeed = 0
    ability.lastDeflector   = nil
    ability.lastDeflectTime = 0
    ability.lastDeflectBall = nil
    ability.lastBallTarget  = nil
    ability.lostTargetAt    = 0
    ability.lastBall        = nil
end

local function detectAbilityEvent(ball, dt)
    if not ball then return nil end
    if not ability.lastDeflector then return nil end
    local age = os.clock() - ability.lastDeflectTime
    if age > ABILITY_SPIKE_T then
        ability.lastDeflector = nil
        return nil
    end
    local spiked, ratio = detectSpike(ball)
    if spiked then
        return {
            type     = string.format("SpeedSpike:%.0f%%:%s", ratio * 100, ability.lastDeflector),
            mod      = 1.30,
            duration = 0.6,
        }
    end
    if ability.lastDir.Magnitude > 1 and dt > 0 then
        local dir = ball.AssemblyLinearVelocity.Unit
        if dir.Magnitude > 0 then
            local dot = math.clamp(dir:Dot(ability.lastDir), -1, 1)
            local deg = math.deg(math.acos(dot))
            if deg > ABILITY_CURVE_D then
                return { type = "Curve:" .. ability.lastDeflector, mod = 1.20, duration = 0.4 }
            end
        end
    end
    return nil
end

local function updateAbility(ball, dt)
    if not CFG.abilityCounter then
        resetAbility()
        return
    end
    local now = os.clock()
    if ball ~= ability.lastBall then
        resetAbility()
        ability.lastBall = ball
    end
    if ball then
        local t = ball:GetAttribute("target")
        if t ~= ability.lastBallTarget then
            onTargetChange(ability.lastBallTarget, t, ball)
            ability.lastBallTarget = t
        end
    end
    if not ball or not isTargeted(ball) then
        if ability.lostTargetAt == 0 then ability.lostTargetAt = now end
        ability.active    = false
        ability.windowMod = 1.0
        if now - ability.lostTargetAt >= ABILITY_LOST_T then
            resetAbility()
        end
        return
    end
    ability.lostTargetAt = 0
    if ability.active and now - ability.startTime > ability.duration then
        ability.active    = false
        ability.windowMod = 1.0
    end
    if now - ability.detectionCooldown < 0.15 then
        local vel = ball.AssemblyLinearVelocity
        if vel.Magnitude > 0.1 then ability.lastDir = vel.Unit end
        return
    end
    local event = detectAbilityEvent(ball, dt)
    if event then
        ability.active          = true
        ability.type            = event.type
        ability.startTime       = now
        ability.duration        = event.duration
        ability.windowMod       = event.mod
        ability.detectionCooldown = now
    end
    local vel = ball.AssemblyLinearVelocity
    if vel.Magnitude > 0.1 then ability.lastDir = vel.Unit end
end

local function abilityWindowMod()
    if not CFG.abilityCounter then return 1.0 end
    if ability.active then return ability.windowMod end
    return 1.0
end

local function attachBall(ball)
    if ballTargetConn then
        ballTargetConn:Disconnect()
        ballTargetConn = nil
    end
    currentBall = ball
    ballLocked  = false
    targetHistory = {}
    ability.lastBall       = ball
    ability.lastBallTarget = ball:GetAttribute("target")
    ballTargetConn = ball:GetAttributeChangedSignal("target"):Connect(function()
        local newT = ball:GetAttribute("target")
        local oldT = ability.lastBallTarget
        pushTargetHistory(newT)
        if oldT ~= newT then
            if oldT == LocalPlayer.Name and CFG.curveDeflectMode ~= "Off" then
                startCamCurve(CFG.curveDeflectMode)
            end
            onTargetChange(oldT, newT, ball)
            ability.lastBallTarget = newT
        end
        if newT == LocalPlayer.Name then
            ballLocked = false
        end
    end)
    local t = ball:GetAttribute("target")
    if t then
        pushTargetHistory(t)
        ability.lastBallTarget = t
    end
    if isTargeted(ball) then
        ballLocked = false
    end
end

        if child:IsA("BasePart") and child:GetAttribute("realBall") == true then
            attachBall(child)
        end
    end))

local Unlock = {
    initialized   = false,
    originals     = {},
    fakeCache     = {},
    allSwordNames = nil,
    swordsByRarity= nil,
}
local FAKE_PREFIX = "tempofake_"

local function isFakeId(id)
    return type(id) == "string" and id:sub(1, #FAKE_PREFIX) == FAKE_PREFIX
end

local function makeFakeId(itemType, name)
    local s = tostring(name)
    if s:sub(1, #FAKE_PREFIX) == FAKE_PREFIX then return s end
    return FAKE_PREFIX .. tostring(itemType) .. "_" .. s
end

local function parseFakeId(id)
    local rest = id:sub(#FAKE_PREFIX + 1)
    local sep  = rest:find("_")
    if not sep then return nil, nil end
    return rest:sub(1, sep - 1), rest:sub(sep + 1)
end

local function unlockEnabled(itemType)
    if itemType == "Sword"     then return CFG.unlockSwords end
    if itemType == "Explosion" then return CFG.unlockExplosions end
    if itemType == "Emote"     then return CFG.unlockEmotes end
    if itemType == "Finisher"  then return CFG.unlockFinishers end
    return false
end

local function fakeItem(id, itemType, name)
    return {
        Name      = name,
        Id        = id,
        ItemType  = itemType,
        TradeLock = { Type = "Permanent", Value = true },
    }
end

function Unlock.enumerateSwords()
    if Unlock.allSwordNames then return Unlock.allSwordNames end
    local list, seen = {}, {}
    local ok, mod = pcall(function()
        return require(ReplicatedStorage.Shared.ReplicatedInstances.Swords)
    end)
    if ok and mod then
        local ok2, col = pcall(function() return mod:GetCollection() end)
        if ok2 and col then
            for name in pairs(col) do
                if not seen[name] then
                    seen[name] = true
                    table.insert(list, name)
                end
            end
        end
    end
    table.sort(list)
    Unlock.allSwordNames = list
    return list
end

function Unlock.groupSwordsByRarity()
    if Unlock.swordsByRarity then return Unlock.swordsByRarity end
    local groups = {}
    for _, r in ipairs(RARITIES) do groups[r] = {} end
    groups["Other"] = {}
    local ok, mod = pcall(function()
        return require(ReplicatedStorage.Shared.ReplicatedInstances.Swords)
    end)
    if not ok or not mod then
        Unlock.swordsByRarity = groups
        return groups
    end
    for _, name in ipairs(Unlock.enumerateSwords()) do
        local sword = nil
        pcall(function() sword = mod:GetSword(name) end)
        local rarity = sword and sword.Rarity
        if rarity and groups[rarity] then
            table.insert(groups[rarity], name)
        else
            table.insert(groups["Other"], name)
        end
    end
    for _, list in pairs(groups) do table.sort(list) end
    Unlock.swordsByRarity = groups
    return groups
end

function Unlock.getSwordsInRarity(rarity)
    if rarity == "All" then return Unlock.enumerateSwords() end
    local groups = Unlock.groupSwordsByRarity()
    return groups[rarity] or {}
end

function Unlock.enumerateCosmetics(itemType)
    local list, seen = {}, {}
    local function add(name)
        if type(name) == "string" and name ~= "" and not seen[name] then
            seen[name] = true
            table.insert(list, name)
        end
    end
    local misc = ReplicatedStorage:FindFirstChild("Misc")
    if itemType == "Sword" then
        for _, n in ipairs(Unlock.enumerateSwords()) do add(n) end
    elseif itemType == "Explosion" then
        if misc then
            local folder = misc:FindFirstChild("DataExplosions")
            if folder then
                for _, c in ipairs(folder:GetChildren()) do add(c.Name) end
            end
        end
    elseif itemType == "Finisher" then
        if misc then
            local folder = misc:FindFirstChild("DataFinishers")
            if folder then
                for _, c in ipairs(folder:GetChildren()) do add(c.Name) end
            end
        end
    elseif itemType == "Ability" then
        if misc then
            local folder = misc:FindFirstChild("DataAbilities")
            if folder then
                for _, c in ipairs(folder:GetChildren()) do add(c.Name) end
            end
        end
    elseif itemType == "Emote" then
        local ok, mod = pcall(function()
            return require(ReplicatedStorage.Shared.ReplicatedInstances.EmoteVFX)
        end)
        if ok and mod then
            local ok2, col = pcall(function() return mod:GetCollection() end)
            if ok2 and col then
                for n in pairs(col) do add(n) end
            end
        end
    end
    return list
end

function Unlock.buildFakeInventory(itemType)
    if Unlock.fakeCache[itemType] then return Unlock.fakeCache[itemType] end
    local inv = {}
    for _, name in ipairs(Unlock.enumerateCosmetics(itemType)) do
        local id = makeFakeId(itemType, name)
        inv[id] = fakeItem(id, itemType, name)
    end
    Unlock.fakeCache[itemType] = inv
    return inv
end

function Unlock.forceEquipSword(name)
    if type(name) ~= "string" or name == "" then
        return false, "Empty sword name"
    end
    local char = LocalPlayer.Character
    if not char then return false, "No character (dead or loading)" end
    local ok, mod = pcall(function()
        return require(ReplicatedStorage.Shared.ReplicatedInstances.Swords)
    end)
    if not ok or not mod then return false, "Swords module load failed" end
    local sword = nil
    pcall(function() sword = mod:GetSword(name) end)
    if not sword then return false, "Sword not in collection: " .. name end
    local ok2, result = pcall(function()
        return mod:EquipSwordTo(char, name, 1, false)
    end)
    if not ok2 then return false, "EquipSwordTo errored: " .. tostring(result) end
    if typeof(result) ~= "Instance" then
        return false, "EquipSwordTo returned " .. typeof(result)
    end
    pcall(function() LocalPlayer:SetAttribute("CurrentlyEquippedSword", name) end)
    pcall(function() char:SetAttribute("CurrentlyEquippedSword", name) end)
    return true, result
end

function Unlock.installOwnershipHooks()
    local shared    = ReplicatedStorage:FindFirstChild("Shared")
    local inv       = shared and shared:FindFirstChild("Inventory")
    local sharedInv = inv and inv:FindFirstChild("Shared")
    if not sharedInv then
        return false, "ReplicatedStorage.Shared.Inventory.Shared not found"
    end
    local ok, mod = pcall(require, sharedInv)
    if not ok or not mod then
        return false, "Failed to require Shared.Inventory.Shared: " .. tostring(mod)
    end
    Unlock.shared = mod
    Unlock.originals.OwnsItem         = mod.OwnsItem
    Unlock.originals.GetItem          = mod.GetItem
    Unlock.originals.FindItems        = mod.FindItems
    Unlock.originals.FindItemsWithKey = mod.FindItemsWithKey

    mod.OwnsItem = function(self, id, ...)
        if isFakeId(id) then return true end
        return Unlock.originals.OwnsItem(self, id, ...)
    end
    mod.GetItem = function(self, id, ...)
        if isFakeId(id) then
            local itemType, name = parseFakeId(id)
            if itemType and name then
                return fakeItem(id, itemType, name)
            end
        end
        return Unlock.originals.GetItem(self, id, ...)
    end
    mod.FindItems = function(self, itemType, ...)
        local real = Unlock.originals.FindItems(self, itemType, ...)
        if not unlockEnabled(itemType) then return real end
        local fake = Unlock.buildFakeInventory(itemType)
        local merged = {}
        if type(real) == "table" then
            for k, v in pairs(real) do merged[k] = v end
        end
        for k, v in pairs(fake) do merged[k] = v end
        return merged
    end
    mod.FindItemsWithKey = function(self, itemType, key, ...)
        local real = Unlock.originals.FindItemsWithKey(self, itemType, key, ...)
        if not unlockEnabled(itemType) then return real end
        local fake = Unlock.buildFakeInventory(itemType)
        local merged = {}
        if type(real) == "table" then
            for k, v in pairs(real) do merged[k] = v end
        end
        for k, v in pairs(fake) do merged[k] = v end
        return merged
    end
    Unlock.initialized = true
    return true, "Hooks installed"
end

function Unlock.init(force)
    if Unlock.initialized and not force then
        return true, "Already initialized"
    end
    return Unlock.installOwnershipHooks()
end

-- ============================================================
local mainConn = nil

local function calcWindow(speed, ping, fps)
    local base = CFG.windowBase
    local maxW = CFG.windowMax
    local minW = CFG.windowMin
    local speedFac = CFG.windowSpeedFac
    local pingFac  = CFG.windowPingFac

    if CFG.autoWindow then
        for _, tier in ipairs(PING_TIERS) do
            if ping * 1000 <= tier.maxPing then
                base     = tier.base
                speedFac = tier.speedFac
                maxW     = tier.max
                minW     = tier.min
                pingFac  = tier.pingFac
                break
            end
        end
    end

    local window = base + speed * speedFac
    if CFG.useAutoPing then
        window = window + ping * pingFac
    end
    window = window + fpsComp(fps)
    window = window * abilityWindowMod()
    if CFG.useDynamicWindow then
        window = window * CFG.simpleRatio
    end
    return math.clamp(window, minW, maxW)
end

local function shouldParry(ball, dt)
    if not ball or not isTargeted(ball) then return false end
    local root = getRoot()
    if not root then return false end

    local pos  = ball.Position
    local vel  = getBallVelocity(ball, dt)
    local speed= vel.Magnitude
    local dist = (pos - root.Position).Magnitude

    if dist > CFG.maxParryDistance then return false end

    if CFG.maxParryAngle < 180 and speed > 1 then
        local toPlayer = (root.Position - pos).Unit
        local dir = vel.Unit
        local ang = math.deg(math.acos(math.clamp(dir:Dot(toPlayer), -1, 1)))
        if ang > CFG.maxParryAngle then return false end
    end

    local now = os.clock()
    if ball == lastParryBall and now - lastParryTime < CFG.sameBallLockout then
        return false
    end

    local accel = 0
    if ballTrack.hasSample and ballTrack.ball == ball then
        accel = (speed - ballTrack.speed) / math.max(dt, 0.001)
    end

    local tti = timeToImpact(dist, speed, accel)
    local ping = getPing()
    local fps  = getFps()
    local window = calcWindow(speed, ping, fps)

    if CFG.curvePrediction and speed > 5 then
        -- slight early bias on high speed
        window = window * 1.05
    end

    if clashActive and CFG.clashPriority then
        local spam = CLASH_MODES[CFG.clashSpamMode] or 0.15
        if now - lastParryTime >= spam then
            return true
        end
    end

    return tti <= window
end

local function mainLoop(dt)
    updateFps(dt)
    if CFG.autoFps then
        autoFpsTier(FPS.smoothed)
    end

    local ball = getRealBall()
    if ball and ball ~= currentBall then
        attachBall(ball)
    end

    if ball then
        local vel = getBallVelocity(ball, dt)
        local speed = vel.Magnitude
        if ballTrack.ball == ball and ballTrack.hasSample then
            ballTrack.accel = (speed - ballTrack.speed) / math.max(dt, 0.001)
        end
        ballTrack.ball      = ball
        ballTrack.pos       = ball.Position
        ballTrack.speed     = speed
        ballTrack.smooth    = ballTrack.smooth + (speed - ballTrack.smooth) * 0.25
        ballTrack.hasSample = true

        updateClash(ball)
        updateAbility(ball, dt)

        if CFG.autoParry and shouldParry(ball, dt) then
            local accuracy = CFG.parryAccuracy
            if accuracy >= 100 or math.random(1, 100) <= accuracy then
                doParry()
                lastParryTime = os.clock()
                lastParryBall = ball
                ballLocked    = true
            end
        end

        -- Orb
        if CFG.orbOn then
            if #orbParts == 0 then createOrb() end
            if orbParts[1] then
                orbParts[1].Position = ball.Position
            end
        else
            destroyOrb()
        end

        -- Overhead
        if CFG.showImpactTime then
            local gui = ensureOverhead()
            if gui then
                local root = getRoot()
                if root then
                    local dist = (ball.Position - root.Position).Magnitude
                    local tti  = timeToImpact(dist, ballTrack.smooth, ballTrack.accel)
                    local label = gui:FindFirstChild("Impact")
                    if label then
                        if tti < 10 then
                            label.Text = string.format("%.0f ms", tti * 1000)
                        else
                            label.Text = ""
                        end
                    end
                end
            end
        else
            destroyOverhead()
        end
    else
        destroyOrb()
        destroyOverhead()
        resetBallTrack()
    end

    -- Movement
    local hum = getHumanoid()
    if hum then
        if CFG.walkEnabled then
            hum.WalkSpeed = CFG.walkSpeed
        end
        if CFG.jumpEnabled then
            hum.JumpPower = CFG.jumpPower
            hum.UseJumpPower = true
        end
    end
end

local function startMainLoop()
    if mainConn then return end
    mainConn = RunService.RenderStepped:Connect(mainLoop)
    track(mainConn)
end

local function stopMainLoop()
    if mainConn then
        pcall(function() mainConn:Disconnect() end)
        mainConn = nil
    end
end

local function refreshLoops()
    if CFG.autoParry or CFG.orbOn or CFG.showImpactTime or CFG.walkEnabled or CFG.jumpEnabled or CFG.autoClash or CFG.abilityCounter then
        startMainLoop()
    else
        -- keep loop light if nothing needs it, but still update fps for auto window
        if not mainConn then startMainLoop() end
    end
end

            end
        end)
    end
end)

local unlockState = {
    rarity     = "Secret",
    searchText = "",
    sword      = nil,
}

local rarityDropdown, swordDropdown

local function refreshSwordList()
    local source = Unlock.getSwordsInRarity(unlockState.rarity)
    local filtered
    if unlockState.searchText and unlockState.searchText ~= "" then
        local q = unlockState.searchText:lower()
        filtered = {}
        for _, n in ipairs(source) do
            if n:lower():find(q, 1, true) then
                table.insert(filtered, n)
            end
        end
    else
        filtered = source
    end
    if #filtered == 0 then
        unlockState.sword = "(no swords match)"
        if swordDropdown and swordDropdown.SetItems then
            pcall(function() swordDropdown:SetItems({ "(no swords match)" }) end)
        end
        return
    end
    local display
    if #filtered > DROPDOWN_CAP then
        display = {}
        for i = 1, DROPDOWN_CAP do display[i] = filtered[i] end
        display[#display + 1] = string.format("(+ %d more — type more in Search)", #filtered - DROPDOWN_CAP)
    else
        display = filtered
    end
    unlockState.sword = display[1]
    if swordDropdown and swordDropdown.SetItems then
        pcall(function() swordDropdown:SetItems(display, display[1]) end)
    elseif swordDropdown and swordDropdown.Refresh then
        pcall(function() swordDropdown:Refresh(display) end)
    end
end
