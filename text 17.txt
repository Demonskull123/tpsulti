local function serialize(v)
    local t = typeof(v)
    if t == "table" then
        local parts = {}
        for k, val in pairs(v) do
            local key
            if typeof(k) == "string" then key = string.format("[%q]", k)
            else key = "[" .. tostring(k) .. "]" end
            table.insert(parts, key .. "=" .. serialize(val))
        end
        return "{" .. table.concat(parts, ",") .. "}"
    elseif t == "string" then
        return string.format("%q", v)
    elseif t == "number" or t == "boolean" then
        return tostring(v)
    elseif t == "Vector3" then
        return string.format("Vector3.new(%0.6f,%0.6f,%0.6f)", v.X, v.Y, v.Z)
    elseif t == "Color3" then
        return string.format("Color3.new(%0.6f,%0.6f,%0.6f)", v.R, v.G, v.B)
    elseif t == "UDim2" then
        return string.format("UDim2.new(%0.6f,%d,%0.6f,%d)", v.X.Scale, v.X.Offset, v.Y.Scale, v.Y.Offset)
    elseif t == "EnumItem" then
        return string.format("Enum.%s.%s", tostring(v.EnumType), v.Name)
    end
    return "nil"
end

local function deserialize(s)
    local fn, err = loadstring("return " .. s, "cfg")
    if not fn then return nil end
    local ok, res = pcall(fn)
    if ok then return res end
    return nil
end

local function deepCopy(v)
    if typeof(v) ~= "table" then return v end
    local out = {}
    for k, val in pairs(v) do out[k] = deepCopy(val) end
    return out
end

local function deepMerge(dst, src)
    for k, v in pairs(src) do
        if typeof(v) == "table" and typeof(dst[k]) == "table" then
            deepMerge(dst[k], v)
        else
            dst[k] = v
        end
    end
end

local ConfigFolder = "TPSULT"
pcall(function() if EXEC and EXEC.makefolder and not (EXEC.isfolder and EXEC.isfolder(ConfigFolder)) then EXEC.makefolder(ConfigFolder) end end)

local function listConfigs()
    local names = {}
    if not EXEC or not EXEC.listfiles then return names end
    local ok, files = pcall(EXEC.listfiles, ConfigFolder)
    if not ok or not files then return names end
    for _, f in ipairs(files) do
        local name = f:match("([^/\\]+)%.json$")
        if name then table.insert(names, name) end
    end
    table.sort(names)
    return names
end

local function saveConfig(name)
    if not EXEC or not EXEC.writefile then return false, "no executor file io" end
    name = tostring(name or ""):gsub("[^%w_%-]", "")
    if name == "" then return false, "bad name" end
    local path = ConfigFolder .. "/" .. name .. ".json"
    local ok, err = pcall(EXEC.writefile, path, serialize(CONFIG))
    if not ok then return false, tostring(err) end
    return true, path
end

local function loadConfig(name)
    if not EXEC or not EXEC.readfile then return false, "no executor file io" end
    name = tostring(name or ""):gsub("[^%w_%-]", "")
    local path = ConfigFolder .. "/" .. name .. ".json"
    if EXEC.isfile and not EXEC.isfile(path) then return false, "not found" end
    local ok, data = pcall(EXEC.readfile, path)
    if not ok or not data then return false, "read failed" end
    local parsed = deserialize(data)
    if typeof(parsed) ~= "table" then return false, "parse failed" end
    deepMerge(CONFIG, parsed)
    return true
end

local function deleteConfig(name)
    if not EXEC then return false end
    name = tostring(name or ""):gsub("[^%w_%-]", "")
    local path = ConfigFolder .. "/" .. name .. ".json"
    if EXEC.isfile and not EXEC.isfile(path) then return false end
    if delfile then pcall(delfile, path) return true end
    if EXEC.writefile then pcall(EXEC.writefile, path, "") return true end
    return false
end

local blockedKeywords = { "exploit", "Banned", "Ban", "Anti", "Detect", "Flag" }

local function checkAndDestroy(script)
    if script:IsA("LocalScript") or script:IsA("Script") then
        local scriptName = script.Name
        local scriptNameLower = scriptName:lower()
        if scriptName == "" or scriptName:match("^%s*$") then
            warn("Destroying script with empty/whitespace name:", script:GetFullName())
            pcall(function() script.Disabled = true script:Destroy() end)
            return true
        end
        for _, keyword in pairs(blockedKeywords) do
            if scriptNameLower:find(keyword:lower()) then
                warn("Destroying blocked script:", scriptName, "Path:", script:GetFullName())
                pcall(function() script.Disabled = true script:Destroy() end)
                return true
            end
        end
    end
    return false
end

        for _, child in ipairs(practice:GetChildren()) do
            if child:IsA("BasePart") and (child.Name == "PSoccerBall" or child.Name == "Ball") then

local function SafeClear(tbl)
    if not tbl then return end
    if table.clear then table.clear(tbl) else for k in pairs(tbl) do tbl[k] = nil end end
end

local function GetReachDistanceForPart(partName)
    if partName == "Head" then
        if not reachSettings.HeadEnabled then return 0 end
        local avg = (reachSettings.ReachX_Head + reachSettings.ReachY_Head + reachSettings.ReachZ_Head) / 3
        return reachSettings.ReachDistance * avg
    elseif partName == "Right Arm" or partName == "Left Arm" then
        local side = partName:match("(%w+)%s")
        if not reachSettings[side .. "ArmEnabled"] then return 0 end
        local avg = (reachSettings.ReachX_Arm + reachSettings.ReachY_Arm + reachSettings.ReachZ_Arm) / 3
        return reachSettings.ReachDistance * avg
    elseif partName == "Right Leg" or partName == "Left Leg" then
        local side = partName:match("(%w+)%s")
        if not reachSettings[side .. "LegEnabled"] then return 0 end
        local avg = (reachSettings.ReachX_Leg + reachSettings.ReachY_Leg + reachSettings.ReachZ_Leg) / 3
        return reachSettings.ReachDistance * avg
    end
    return 0
end

local function CreateLimbHitbox(bodyPart, partName)
    local hitbox = Instance.new("Part")
    hitbox.Name = partName .. "_Hitbox"
    hitbox.Anchored = false
    hitbox.CanCollide = false
    hitbox.Transparency = 1
    hitbox.Massless = true
    if partName == "Head" then
        hitbox.Size = Vector3.new(bodyPart.Size.X*reachSettings.ReachX_Head*2, bodyPart.Size.Y*reachSettings.ReachY_Head*2, bodyPart.Size.Z*reachSettings.ReachZ_Head*2)
    elseif partName == "Right Arm" or partName == "Left Arm" then
        hitbox.Size = Vector3.new(bodyPart.Size.X*reachSettings.ReachX_Arm*2, bodyPart.Size.Y*reachSettings.ReachY_Arm*2, bodyPart.Size.Z*reachSettings.ReachZ_Arm*2)
    else
        hitbox.Size = Vector3.new(bodyPart.Size.X*reachSettings.ReachX_Leg*2, bodyPart.Size.Y*reachSettings.ReachY_Leg*2, bodyPart.Size.Z*reachSettings.ReachZ_Leg*2)
    end
    hitbox.CFrame = bodyPart.CFrame
    hitbox.Parent = bodyPart
    local weld = Instance.new("WeldConstraint")
    weld.Part0 = bodyPart
    weld.Part1 = hitbox
    weld.Parent = hitbox
    return hitbox
end

local function GetVisTransparencyFor(partName)
    if partName == "Head" then return reachSettings.HeadVisTransparency
    elseif partName == "Right Arm" or partName == "Left Arm" then return reachSettings.ArmVisTransparency
    else return reachSettings.LegVisTransparency end
end

local function UpdateLimbVisualizer(partName)
    local char = player.Character
    if not char then return end
    local bodyPart = char:FindFirstChild(partName)
    if not bodyPart then return end
    local reachDist = GetReachDistanceForPart(partName)
    if reachDist > 0 then
        if LimbHitboxes[partName] then LimbHitboxes[partName]:Destroy() end
        LimbHitboxes[partName] = CreateLimbHitbox(bodyPart, partName)
    else
        if LimbHitboxes[partName] then LimbHitboxes[partName]:Destroy() LimbHitboxes[partName] = nil end
    end
    local visOn = false
    if partName == "Head" then visOn = reachSettings.HeadVisualizer
    elseif partName == "Right Arm" then visOn = reachSettings.RightArmVisualizer
    elseif partName == "Left Arm" then visOn = reachSettings.LeftArmVisualizer
    elseif partName == "Right Leg" then visOn = reachSettings.RightLegVisualizer
    elseif partName == "Left Leg" then visOn = reachSettings.LeftLegVisualizer end
    if LimbVisualizers[partName] then LimbVisualizers[partName]:Destroy() LimbVisualizers[partName] = nil end
    if visOn and LimbHitboxes[partName] then
        local hitbox = LimbHitboxes[partName]
        local vis = Instance.new("Part")
        vis.Name = partName .. "_Visualizer"
        vis.Size = hitbox.Size
        vis.CFrame = hitbox.CFrame
        vis.Anchored = false
        vis.CanCollide = false
        vis.Transparency = GetVisTransparencyFor(partName)
        vis.Color = Color3.fromRGB(0, 255, 100)
        vis.Material = Enum.Material.Neon
        vis.Massless = true
        vis.Parent = hitbox
        local w = Instance.new("WeldConstraint")
        w.Part0 = hitbox
        w.Part1 = vis
        w.Parent = vis
        LimbVisualizers[partName] = vis
    end
end

local function RefreshVisTransparency()
    for partName, v in pairs(LimbVisualizers) do
        if v and v.Parent then v.Transparency = GetVisTransparencyFor(partName) end
    end
end

local function FindBallPartLimb(part)
    if part:IsA("BasePart") and (part.Name == "TPS" or part.Name == "PSoccerBall" or BALL_SET[part.Name]) then return part end
    local current = part
    while current and current ~= Workspace do
        if current:IsA("BasePart") and (current.Name == "PSoccerBall" or current.Name == "TPS" or BALL_SET[current.Name]) then return current end
        current = current.Parent
    end
    return nil
end

local function CreateAllLimbHitboxes()
    local char = player.Character
    if not char then return end
    for _, partName in ipairs(LimbBodyParts) do
        local bodyPart = char:FindFirstChild(partName)
        if bodyPart and GetReachDistanceForPart(partName) > 0 then
            if LimbHitboxes[partName] then LimbHitboxes[partName]:Destroy() end
            LimbHitboxes[partName] = CreateLimbHitbox(bodyPart, partName)
            UpdateLimbVisualizer(partName)
        end
    end
end

local function RemoveAllLimbHitboxes()
    for _, h in pairs(LimbHitboxes) do if h then h:Destroy() end end
    for _, v in pairs(LimbVisualizers) do if v then v:Destroy() end end
    SafeClear(LimbHitboxes)
    SafeClear(LimbVisualizers)
    SafeClear(LimbActiveTouches)
end

local function CheckLimbBallCollisions()
    local char = player.Character
    if not char or not char.Parent then return end
    local currentTime = tick()
    local newTouches = {}
    for partName, hitbox in pairs(LimbHitboxes) do
        if hitbox and hitbox.Parent then
            local bodyPart = char:FindFirstChild(partName)
            if bodyPart then
                local op = OverlapParams.new()
                op.FilterDescendantsInstances = {char}
                op.FilterType = Enum.RaycastFilterType.Exclude
                local partsInBox = Workspace:GetPartBoundsInBox(hitbox.CFrame, hitbox.Size, op)
                for _, part in ipairs(partsInBox) do
                    local ballPart = FindBallPartLimb(part)
                    if ballPart then
                        local key = partName .. ":" .. ballPart:GetFullName()
                        newTouches[key] = {part = ballPart, bodyPart = bodyPart, time = currentTime}
                        if not LimbActiveTouches[key] or (currentTime - LimbActiveTouches[key].time) > LIMB_TOUCH_COOLDOWN then
                            LimbActiveTouches[key] = {part = ballPart, bodyPart = bodyPart, time = currentTime}
                            pcall(function()
                                firetouchinterest(ballPart, bodyPart, 0)
                                task.wait()
                                firetouchinterest(ballPart, bodyPart, 1)
                            end)
                        end
                    end
                end
            end
        end
    end
    for key, d in pairs(LimbActiveTouches) do
        if not newTouches[key] and (currentTime - d.time) > LIMB_TOUCH_COOLDOWN then LimbActiveTouches[key] = nil end
    end
end

    local now = tick()
    if now - limbLastUpdate >= 0.1 then
        limbLastUpdate = now
        pcall(CheckLimbBallCollisions)
    end
end)

-- Predicted ball position using velocity + gravity over dt
local function predictBallPos(ball, dt)
    local v = ball.AssemblyLinearVelocity
    return ball.Position + v * dt + Vector3.new(0, -Workspace.Gravity * 0.5 * dt * dt, 0)
end

-- Closest approach of ball to root within lookahead; returns timeToClosest, willPassNear
local function timeToClosestApproach(ballPos, ballVel, rootPos, lookahead)
    local relPos = rootPos - ballPos
    local relVel = -ballVel
    local vv = relVel:Dot(relVel)
    if vv < 1e-4 then return lookahead, (ballPos - rootPos).Magnitude end
    local t = math.clamp(relPos:Dot(relVel) / vv, 0, lookahead)
    local closestPos = ballPos + ballVel * t + Vector3.new(0, -Workspace.Gravity * 0.5 * t * t, 0)
    return t, (closestPos - rootPos).Magnitude
end

local function AutoCatch_ArmsFirst(ball, char)
    for _, name in ipairs({"Right Arm", "Left Arm"}) do
        local limb = char:FindFirstChild(name)
        if limb then AutoCatch_TouchLimb(limb, ball, AUTO_CATCH_ARM_FIRES) end
    end
    for _, name in ipairs({"Right Leg", "Left Leg", "Head"}) do
        local limb = char:FindFirstChild(name)
        if limb then AutoCatch_TouchLimb(limb, ball, AUTO_CATCH_BODY_FIRES) end
    end
end

local function AutoCatch_KillPushback(ball, root)
    -- zero any ball velocity component pointing away from root when contact is imminent.
    local toRoot = (root.Position - ball.Position)
    local dist = toRoot.Magnitude
    if dist < 1e-3 then return end
    local dir = toRoot.Unit
    local vel = ball.AssemblyLinearVelocity
    local away = vel:Dot(-dir)
    if away > 0 then
        -- remove only the away-facing component; preserve tangential so ball still rolls
        local tangential = vel + dir * away
        ball.AssemblyLinearVelocity = tangential
    end
end

local function AutoCatch_Tick()
    if not AutoCatch.Enabled then return end
    local char = player.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local balls = {}
    for _, b in ipairs(TrackedBalls) do
        if b and b.Parent then table.insert(balls, b) end
    end
    local main = GetMainBall()
    if main and not table.find(balls, main) then table.insert(balls, main) end
    for _, b in ipairs(GetPracticeBalls()) do
        if not table.find(balls, b) then table.insert(balls, b) end
    end
    if #balls == 0 then return end

    local now = os.clock()
    local rootPos = root.Position

    for _, ball in ipairs(balls) do
        local dist = (ball.Position - rootPos).Magnitude
        if dist <= AutoCatch.Distance then
            local ballVel = ball.AssemblyLinearVelocity
            local tt, closest = timeToClosestApproach(ball.Position, ballVel, rootPos, 0.6)
            local imminent = dist <= AutoCatch.StickyRange or (tt < 0.35 and closest <= AutoCatch.StickyRange)

            if dist <= AutoCatch.StickyRange then
                AutoCatch.StickyUntil = now + AutoCatch.StickyHold
                AutoCatch.StickyBall = ball
            end

            AutoCatch_ArmsFirst(ball, char)

            -- kill away-facing velocity whenever the ball is close enough to push
            if dist <= (AutoCatch.StickyRange + 1.0) then
                AutoCatch_KillPushback(ball, root)
            end

            if AutoCatch.Force then
                -- pull only when ball is outside stop radius; never push past centerline
                if dist > AutoCatch.StopRadius then
                    local toRoot = (rootPos - ball.Position)
                    local dir = toRoot.Unit
                    local alongSpeed = ball.AssemblyLinearVelocity:Dot(dir)
                    if alongSpeed < AutoCatch.PullStrength then
                        local need = math.min(AutoCatch.PullStrength - alongSpeed, 40)
                        ball.AssemblyLinearVelocity = ball.AssemblyLinearVelocity + dir * need
                    end
                else
                    -- inside stop radius: bleed all velocity so it doesn't shoot through
                    ball.AssemblyLinearVelocity = ball.AssemblyLinearVelocity * 0.4
                end
            end

            -- sticky phase: hold ball near root without launching it past
            if now < AutoCatch.StickyUntil and AutoCatch.StickyBall == ball then
                local stickyTarget = rootPos + (ball.Position - rootPos).Unit * math.max(AutoCatch.StopRadius, 0.6)
                if stickyTarget.Magnitude ~= stickyTarget.Magnitude then stickyTarget = rootPos end
                local toTarget = stickyTarget - ball.Position
                ball.AssemblyLinearVelocity = ball.AssemblyLinearVelocity * 0.6 + toTarget * 8
            end

            if AutoCatch.Dampen then
                pcall(function() ball.AssemblyAngularVelocity = ball.AssemblyAngularVelocity * 0.5 end)
            end
        end
    end
end

local function ccGetModule(tool, which)
    if not tool then return nil end
    local m = tool:FindFirstChild(which)
    if m then return m end

        if c.Name:lower() == which:lower() then return c end
    end
    return nil

local function ccJump()
    local mgr = ReplicatedStorage:FindFirstChild("Managers")
    local bm = mgr and mgr:FindFirstChild("BManager")
    local ani = bm and bm:FindFirstChild("Animate")
    local jmp = ani and ani:FindFirstChild("jump")
    if jmp then
        local ok = pcall(function()
            local mod = require(jmp)
            if type(mod) == "function" then mod()
            elseif type(mod) == "table" and type(mod.Play) == "function" then mod.Play()
            elseif type(mod) == "table" and type(mod.Jump) == "function" then mod.Jump() end
        end)
        if ok then return true end
    end
    local char = player.Character
    local h = char and char:FindFirstChildOfClass("Humanoid")
    if h and h.Parent and h.Health > 0 then
        pcall(function()
            h:SetStateEnabled(Enum.HumanoidStateType.Jumping, true)
            h:SetStateEnabled(Enum.HumanoidStateType.Freefall, true)
            h.Jump = true
            h:ChangeState(Enum.HumanoidStateType.Jumping)
        end)
        return true
    end
    return false
end

    stroke.Thickness = 1.4

    stroke.Parent = btn
    local dragging, dragStart, startPos, fired = false, nil, nil, false
    btn.InputBegan:Connect(function(input)
        if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then return end
        if input.UserInputState and input.UserInputState ~= Enum.UserInputState.Begin then return end
        if CC.btn.locked then
            if CC.mobEnabled then ccRunCombo(def.combo) end
            fired = true
            return
        end
        dragging = true
        dragStart = input.Position
        startPos = btn.Position
        fired = false
    end)
    btn.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            local d = input.Position - dragStart
            btn.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)

        if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then return end

            CC.positions[def.id] = btn.Position
            if (input.Position - (dragStart or input.Position)).Magnitude < 6 and not fired and CC.mobEnabled then
                ccRunCombo(def.combo)
            end
        end
        fired = false
    end)
    ccRefs[def.id] = { btn = btn, stroke = stroke }
