local blockedKw = { "exploit","Banned","Ban","Anti","Detect","Flag" }
local function checkKill(s)
    if not (s:IsA("LocalScript") or s:IsA("Script")) then return end
    local n = s.Name
    if n == "" or n:match("^%s*$") then pcall(function() s.Disabled = true; s:Destroy() end) return end
    local ln = n:lower()
    for _, k in ipairs(blockedKw) do
        if ln:find(k:lower()) then pcall(function() s.Disabled = true; s:Destroy() end) return end
    end
end

local reach = {
    HeadEnabled = false, HeadVis = false, RXH = 1, RYH = 1, RZH = 1,
    RAEnabled  = false, RAVis  = false, RXA = 1, RYA = 1, RZA = 1,
    LAEnabled  = false, LAVis  = false,
    RLEnabled  = false, RLVis  = false, RXL = 1, RYL = 1, RZL = 1,
    LLEEnabled = false, LLVis  = false,
    Dist = 50,
}
local BodyParts   = { "Head","Right Arm","Left Arm","Right Leg","Left Leg" }
local Character   = player.Character or player.CharacterAdded:Wait()
local Hitboxes    = {}
local Visualizers = {}
local ActiveTouches = {}
local TOUCH_COOLDOWN = 0.1

local function reachDist(partName)
    if partName == "Head" then
        if not reach.HeadEnabled then return 0 end
        return reach.Dist * ((reach.RXH + reach.RYH + reach.RZH) / 3)
    elseif partName == "Right Arm" or partName == "Left Arm" then
        local side = partName:match("(%w+)%s")
        if not reach[side.."AEnabled"] then return 0 end
        return reach.Dist * ((reach.RXA + reach.RYA + reach.RZA) / 3)
    elseif partName == "Right Leg" or partName == "Left Leg" then
        local side = partName:match("(%w+)%s")
        if not reach[side.."LEnabled"] then return 0 end
        return reach.Dist * ((reach.RXL + reach.RYL + reach.RZL) / 3)
    end
    return 0
end

local function mkHitbox(bp, name)
    local hb = Instance.new("Part")
    hb.Name = name.."_Hitbox"
    hb.Anchored = false; hb.CanCollide = false; hb.Transparency = 1; hb.Massless = true
    if name == "Head" then
        hb.Size = Vector3.new(bp.Size.X*reach.RXH*2, bp.Size.Y*reach.RYH*2, bp.Size.Z*reach.RZH*2)
    elseif name == "Right Arm" or name == "Left Arm" then
        hb.Size = Vector3.new(bp.Size.X*reach.RXA*2, bp.Size.Y*reach.RYA*2, bp.Size.Z*reach.RZA*2)
    else
        hb.Size = Vector3.new(bp.Size.X*reach.RXL*2, bp.Size.Y*reach.RYL*2, bp.Size.Z*reach.RZL*2)
    end
    hb.CFrame = bp.CFrame
    hb.Parent = bp
    local w = Instance.new("WeldConstraint"); w.Part0 = bp; w.Part1 = hb; w.Parent = hb
    return hb
end

local function updVis(name)
    local bp = Character:FindFirstChild(name); if not bp then return end
    if Hitboxes[name] then Hitboxes[name]:Destroy(); Hitboxes[name] = nil end
    if reachDist(name) > 0 then Hitboxes[name] = mkHitbox(bp, name) end
    if Visualizers[name] then Visualizers[name]:Destroy(); Visualizers[name] = nil end
    local show = false
    if name == "Head"      then show = reach.HeadVis
    elseif name == "Right Arm" then show = reach.RAVis
    elseif name == "Left Arm"  then show = reach.LAVis
    elseif name == "Right Leg" then show = reach.RLVis
    elseif name == "Left Leg"  then show = reach.LLVis end
    if show and Hitboxes[name] then
        local hb = Hitboxes[name]
        local v = Instance.new("Part")
        v.Name = name.."_Visualizer"; v.Size = hb.Size; v.CFrame = hb.CFrame
        v.Anchored = false; v.CanCollide = false; v.Transparency = 0.7
        v.Color = Color3.fromRGB(0,255,0); v.Material = Enum.Material.Neon; v.Massless = true
        v.Parent = hb
        local w = Instance.new("WeldConstraint"); w.Part0 = hb; w.Part1 = v; w.Parent = v
        Visualizers[name] = v
    end
end

local function findBallPart(p)
    if p:IsA("BasePart") and (p.Name == "TPS" or p.Name == "PSoccerBall") then return p end
    local c = p
    while c and c ~= workspace do
        if c:IsA("BasePart") and c.Name == "PSoccerBall" then return c end
        c = c.Parent
    end
end

local function mkAllHitboxes()
    for _, n in ipairs(BodyParts) do
        local bp = Character:FindFirstChild(n)
        if bp and reachDist(n) > 0 then
            if Hitboxes[n] then Hitboxes[n]:Destroy() end
            Hitboxes[n] = mkHitbox(bp, n)
        end
    end
end

local function rmAllHitboxes()
    for _, h in pairs(Hitboxes)    do if h then h:Destroy() end end
    for _, v in pairs(Visualizers) do if v then v:Destroy() end end
    table.clear(Hitboxes); table.clear(Visualizers); table.clear(ActiveTouches)
end

local function checkBallCollisions()
    if not Character or not Character.Parent then return end
    local now = tick(); local newT = {}
    for name, hb in pairs(Hitboxes) do
        if hb and hb.Parent then
            local bp = Character:FindFirstChild(name)
            if bp then
                local op = OverlapParams.new()
                op.FilterDescendantsInstances = { Character }
                op.FilterType = Enum.RaycastFilterType.Exclude
                for _, p in ipairs(workspace:GetPartBoundsInBox(hb.CFrame, hb.Size, op)) do
                    local ball = findBallPart(p)
                    if ball then
                        local key = name..":"..ball:GetFullName()
                        newT[key] = true
                        if not ActiveTouches[key] or (now - ActiveTouches[key]) > TOUCH_COOLDOWN then
                            ActiveTouches[key] = now
                            pcall(function()
                                firetouchinterest(ball, bp, 0); task.wait()
                                firetouchinterest(ball, bp, 1)
                            end)
                        end
                    end
                end
            end
        end
    end
    for k, t in pairs(ActiveTouches) do
        if not newT[k] and (now - t) > TOUCH_COOLDOWN then ActiveTouches[k] = nil end
    end
end

mkAllHitboxes()
local lastUpd = 0
RunService.Heartbeat:Connect(function()
    local now = tick()
    if now - lastUpd >= 0.1 then lastUpd = now; pcall(checkBallCollisions) end
end)
player.CharacterAdded:Connect(function(c)
    Character = c; rmAllHitboxes(); task.wait(2); mkAllHitboxes()
end)

    InfStamEnabled = not InfStamEnabled
    if not InfStamEnabled then notify("Stam OFF") return end

    local mt = getrawmetatable(game); setreadonly(mt, false)
    local old = mt.__namecall
    local intercepted, disabled, returns = {}, {}, {}
    local ncm = getnamecallmethod or get_namecall_method
    local methods = { RemoteEvent = "FireServer", RemoteFunction = "InvokeServer", BindableEvent = "Fire", BindableFunction = "Invoke" }

    getgenv().InterceptRemoteArgs = function(e, a) intercepted[e] = a end
    getgenv().SpoofReturn         = function(f, r) returns[f] = r end
    getgenv().DisableRemote       = function(e) disabled[e] = true end
    getgenv().EnableRemote        = function(e) disabled[e] = false end

    mt.__namecall = newcclosure(function(self, ...)
        if not InfStamEnabled then return old(self, ...) end
        local args = {...}; local ev = ncm()
        if methods[self.ClassName] == ev then
            if disabled[self] then return
            elseif intercepted[self] then
                local i = intercepted[self]
                if typeof(i) == "function" then args = { i(unpack(args)) }
                elseif typeof(i) == "table"  then args = i end
            end
        end
        local ret = { old(self, unpack(args)) }
        if returns[self] and ret then ret = { returns[self](unpack(ret)) } end
        return unpack(ret)
    end)
    local oi = mt.__index
    mt.__index = function(o, k)
        if tostring(o) == "Humanoid" and tostring(k) == "WalkSpeed" then return 16 end
        return oi(o, k)
    end
    setreadonly(mt, true)

    local ch = player.Character or player.CharacterAdded:Wait()
    local hum = ch:WaitForChild("Humanoid")
    local RS = game:GetService("ReplicatedStorage")
    if RS:FindFirstChild("FE") and RS.FE:FindFirstChild("Sprint") then
        InterceptRemoteArgs(RS.FE.Sprint, function() return "Ended" end)
    end
    hum.WalkSpeed = 22
    local mouse = player:GetMouse()
    mouse.KeyDown:Connect(function(k)
        if not InfStamEnabled then return end
        if k:lower() == "r" then hum.WalkSpeed = 16 end
    end)
    mouse.Button1Down:Connect(function()
        if InfStamEnabled then hum.WalkSpeed = 22 end
    end)
    local pg = player.PlayerGui:FindFirstChild("Start")
    if pg and pg:FindFirstChild("Points") then
        local is = pg.Points:FindFirstChild("InstantStamina")
        if is then is.Text = "lov active" end
    end

    curveFolder = Instance.new("Folder"); curveFolder.Name = "BallPredictionCurve"; curveFolder.Parent = workspace

    local segs = {}
    for i = 1, cfg.segs do
        local s = Instance.new("Part")
        s.Name = "S_"..i; s.Anchored = true; s.CanCollide = false; s.CastShadow = false
        s.Material = Enum.Material.Neon; s.Transparency = cfg.segT
        s.Size = cfg.segSize; s.Color = Color3.fromRGB(0,255,0); s.Parent = curveFolder
        segs[i] = s
    end
    local mk = Instance.new("Part")
    mk.Name = "LandingMarker"; mk.Anchored = true; mk.CanCollide = false; mk.CastShadow = false
    mk.Material = Enum.Material.Neon; mk.Color = Color3.fromRGB(0,255,0); mk.Transparency = 0.4
    mk.Size = Vector3.new(4,0.2,4); mk.Shape = Enum.PartType.Cylinder; mk.Parent = curveFolder

    local function sim(sp, sv)
        local pts = {}
        local pos, vel = sp, sv
        local t, b = 0, 0
        local n = 0
        local air = cfg.air ^ cfg.dt
        while t < cfg.maxT and b < 8 and n < cfg.maxPts do
            t = t + cfg.dt; n = n + 1
            vel = (vel + gV * cfg.dt) * air
            if vel.Y < -cfg.term then vel = Vector3.new(vel.X, -cfg.term, vel.Z) end
            pos = pos + vel * cfg.dt
            pts[n] = { pos, vel }
            if pos.Y <= cfg.ground and vel.Y < 0 then
                pos = Vector3.new(pos.X, cfg.ground, pos.Z)
                local by = -vel.Y * cfg.damp
                vel = Vector3.new(vel.X * cfg.fric, by, vel.Z * cfg.fric)
                b = b + 1
                if vel.Magnitude < cfg.minB then pts[n] = { pos, Vector3.zero }; break end
            end
            if pos.Y < -500 or vel.Magnitude < 0.3 then break end
        end
        return pts, n
    end

    local frame = 0
    predConn = RunService.Heartbeat:Connect(function()
        frame = frame + 1
        if frame % cfg.updRate ~= 0 then return end
        local ball = getBall()
        if not ball then
            for i = 1, cfg.segs do segs[i].Transparency = 1 end
            mk.Transparency = 1; return
        end
        local v = ball.AssemblyLinearVelocity
        if v.Magnitude < cfg.minV then
            for i = 1, cfg.segs do segs[i].Transparency = 1 end
            mk.Transparency = 1; return
        end
        local pts, n = sim(ball.Position, v)
        if n > 1 then
            local step = (n - 1) / (cfg.segs - 1)
            for i = 1, cfg.segs do
                local pi = math.floor((i - 1) * step) + 1
                local ni = math.min(pi + 1, n)
                local cp = pts[pi] and pts[pi][1]
                local np = pts[ni] and pts[ni][1]
                if cp and np then
                    local mid = (cp + np) * 0.5
                    local d = (np - cp).Magnitude
                    if d > 0.01 then
                        segs[i].Size = Vector3.new(0.3, 0.3, d)
                        segs[i].CFrame = CFrame.lookAt(mid, mid + (np - cp) / d)
                        segs[i].Transparency = cfg.segT
                    else segs[i].Transparency = 1 end
                else segs[i].Transparency = 1 end
            end
        end
        for i = 2, n do
            local cp, cv = pts[i][1], pts[i][2]
            local pp = pts[i-1][1]
            if cp.Y <= 0.5 and pp.Y > 0.5 and cv.Y < 0 then
                mk.CFrame = CFrame.new(cp + Vector3.new(0,0.1,0)) * CFrame.Angles(0,0,math.pi*0.5)
                mk.Transparency = 0.4; break
            else mk.Transparency = 1 end
        end
    end)
    notify("Predictor ON")
end)

    local UIS = game:GetService("UserInputService")
    local plr = Players.LocalPlayer
    local char = plr.Character or plr.CharacterAdded:Wait()
    local hrp = char:WaitForChild("HumanoidRootPart")
    local hum = char:WaitForChild("Humanoid")

    local sg = Instance.new("ScreenGui"); sg.Name = "AimbotGui"; sg.ResetOnSpawn = false; sg.Parent = plr.PlayerGui
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0, 120, 0, 50); btn.Position = UDim2.new(0.5, -60, 0.1, 0)
    btn.Text = "Aimbot: OFF"; btn.BackgroundColor3 = Color3.fromRGB(255,0,0)
    btn.TextColor3 = Color3.new(1,1,1); btn.Font = Enum.Font.GothamBold; btn.TextSize = 16; btn.Parent = sg

    local aimOn, mouseDown = false, false
    btn.MouseButton1Click:Connect(function()
        aimOn = not aimOn
        btn.Text = "Aimbot: "..(aimOn and "ON" or "OFF")
        btn.BackgroundColor3 = aimOn and Color3.fromRGB(0,255,0) or Color3.fromRGB(255,0,0)
    end)
    UIS.InputBegan:Connect(function(i, gp) if not gp and i.UserInputType == Enum.UserInputType.MouseButton1 then mouseDown = true end end)
    UIS.InputEnded:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 then mouseDown = false end end)

    local function closestBall()
        local b, d = nil, math.huge
        local tp = workspace:FindFirstChild("TPSSystem") and workspace.TPSSystem:FindFirstChild("TPS")
        if tp then local dd = (tp.Position - hrp.Position).Magnitude; if dd < d then d = dd; b = tp end end
        local pr = workspace:FindFirstChild("Practice")
        if pr then
            for _, o in ipairs(pr:GetDescendants()) do
                if o.Name == "PSoccerBall" and o:IsA("BasePart") then
                    local dd = (o.Position - hrp.Position).Magnitude
                    if dd < d then d = dd; b = o end
                end
            end
        end
        return b, d
    end

    local function goal()
        local t = workspace.TPSSystem
        local d1 = (t.Part1.Position - hrp.Position).Magnitude
        local d2 = (t.Part2.Position - hrp.Position).Magnitude
        return d1 < d2 and t.Part1 or t.Part2
    end

    local function aimPoint(g, bp)
        local gp, gs = g.Position, g.Size
        local side = bp.X > gp.X and "right" or "left"
        local toGoal = (gp - bp).Magnitude
        local ang = math.deg(math.atan2(math.abs(bp.X - gp.X), math.abs(bp.Z - gp.Z)))
        local m = CFG.postMargin
        if toGoal > 35 then m = m * 1.2 elseif toGoal < 15 then m = m * 1.3 end
        if ang > 50 then m = m * 1.8 elseif ang > 35 then m = m * 1.4 end
        local xo = (gs.X / 2) - m
        if side == "left" then xo = -xo end
        if toGoal > 35 then xo = xo * 0.55 elseif toGoal > 28 then xo = xo * 0.65 elseif toGoal < 10 then xo = xo * 0.88 end
        local ap = gp + Vector3.new(xo, gs.Y / 2 - 0.7, 0)
        local dm = (toGoal > 40 and 3.5) or (toGoal > 30 and 2.3) or (toGoal > 20 and 1.5) or 1.0
        local am = (ang > 65 and 2.4) or (ang > 45 and 1.7) or 1.0
        local cd = side == "left" and 1 or -1
        return ap + Vector3.new(CFG.curve * dm * am * cd, 0, 0)
    end

    local origRot = hum.AutoRotate
    RunService.RenderStepped:Connect(function()
        if not aimOn then hum.AutoRotate = origRot; return end
        local b, d = closestBall()
        if not b or d >= CFG.maxDist or not mouseDown then hum.AutoRotate = origRot; return end
        local g = goal(); if not g then return end
        local ap = aimPoint(g, b.Position)
        local look = Vector3.new(ap.X, hrp.Position.Y, ap.Z)
        local tc = CFrame.new(hrp.Position, look)
        local s = math.clamp(CFG.smooth * (d < 8 and 1.8 or d > 25 and 0.85 or 1), CFG.minSmooth, 0.95)
        hum.AutoRotate = false
        hrp.CFrame = hrp.CFrame:Lerp(tc, s)
    end)
    plr.CharacterAdded:Connect(function(c)
        hrp = c:WaitForChild("HumanoidRootPart"); hum = c:WaitForChild("Humanoid")
        origRot = hum.AutoRotate
    end)
    notify("Aimbot loaded")
end)

local function ccJump()
    local RS = game:GetService("ReplicatedStorage")
    local bm = RS:FindFirstChild("Managers") and RS.Managers:FindFirstChild("BManager")
    local ani = bm and bm:FindFirstChild("Animate"); local j = ani and ani:FindFirstChild("jump")
    if j then
        local ok = pcall(function()
            local m = require(j)
            if type(m) == "function" then m()
            elseif type(m) == "table" and type(m.Play) == "function" then m.Play()
            elseif type(m) == "table" and type(m.Jump) == "function" then m.Jump() end
        end)
        if ok then return true end
    end
    local h = ccHum()
    if h and h.Parent and h.Health > 0 then
        pcall(function()
            h:SetStateEnabled(Enum.HumanoidStateType.Jumping, true)
            h:SetStateEnabled(Enum.HumanoidStateType.Freefall, true)
            h.Jump = true; h:ChangeState(Enum.HumanoidStateType.Jumping)
        end)
        return true
    end
    return false
end

local function ccBuildPayload()
    local p = {}
    for id, ud in pairs(CC.positions) do p[id] = { xs=ud.X.Scale, xo=ud.X.Offset, ys=ud.Y.Scale, yo=ud.Y.Offset } end
    return { v = 7, btn = { size = CC.btn.size, transparency = CC.btn.transparency, locked = CC.btn.locked }, positions = p, timings = { keepToSave = CC.keepToSave } }
end
local function ccApplyPayload(d)
    if type(d) ~= "table" then return end
    if type(d.btn) == "table" then
        if type(d.btn.size)         == "number"  then CC.btn.size         = d.btn.size end
        if type(d.btn.transparency) == "number"  then CC.btn.transparency = d.btn.transparency end
        if type(d.btn.locked)       == "boolean" then CC.btn.locked       = d.btn.locked end
    end
    if type(d.positions) == "table" then
        for id, p in pairs(d.positions) do
            if type(p) == "table" and p.xs and p.xo and p.ys and p.yo then CC.positions[id] = UDim2.new(p.xs, p.xo, p.ys, p.yo) end
        end
    end
    if type(d.timings) == "table" then for k, v in pairs(d.timings) do if CC[k] ~= nil and type(v) == type(CC[k]) then CC[k] = v end end end
end
local function ccSaveState() if not ccCanSave then return end; pcall(function() writefile(CC_FILE, Http:JSONEncode(ccBuildPayload())) end) end
CC_API.SaveState = ccSaveState
local function ccLoadState()
    if not ccCanSave then return end
    local ex = false; pcall(function() ex = isfile(CC_FILE) end); if not ex then return end
    local ok, d = pcall(function() return Http:JSONDecode(readfile(CC_FILE)) end)
    if ok then ccApplyPayload(d) end
end

local function ccAddButton(def)
    local pos = CC.positions[def.id] or CC_DEFAULT_POS[def.id] or UDim2.new(0.5,0,0.5,0)
    CC.positions[def.id] = pos
    local b = Instance.new("TextButton")
    b.Name = def.id; b.Size = UDim2.new(0, CC.btn.size, 0, CC.btn.size); b.Position = pos
    b.BackgroundColor3 = Color3.fromRGB(22,22,28)
    b.BackgroundTransparency = CC.btn.transparency; b.TextColor3 = Color3.fromRGB(0,255,130)
    b.TextTransparency = CC.btn.transparency
    b.Font = Enum.Font.GothamBold; b.TextSize = 14; b.Text = def.label or def.id
    b.BorderSizePixel = 0; b.Active = true; b.Visible = true; b.AutoButtonColor = false; b.Parent = ccGui
    local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0,10); c.Parent = b
    local s = Instance.new("UIStroke")
    s.Color = CC.btn.locked and Color3.fromRGB(0,255,130) or Color3.fromRGB(255,200,60)
    s.Thickness = 1.4; s.Transparency = math.clamp(0.35 + CC.btn.transparency, 0, 1); s.Parent = b

    local dragging, dragStart, startPos, fired = false, nil, nil, false
    b.InputBegan:Connect(function(i)
        if i.UserInputType ~= Enum.UserInputType.MouseButton1 and i.UserInputType ~= Enum.UserInputType.Touch then return end
        if i.UserInputState and i.UserInputState ~= Enum.UserInputState.Begin then return end
        if CC.btn.locked then if CC.mobEnabled then ccRunCombo(def.combo) end; fired = true; return end
        dragging = true; dragStart = i.Position; startPos = b.Position; fired = false
    end)
    b.InputChanged:Connect(function(i)
        if not dragging then return end
        if i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch then
            local d = i.Position - dragStart
            b.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
    b.InputEnded:Connect(function(i)
        if i.UserInputType ~= Enum.UserInputType.MouseButton1 and i.UserInputType ~= Enum.UserInputType.Touch then return end
        if dragging then
            dragging = false; CC.positions[def.id] = b.Position; ccSaveState()
            if (i.Position - (dragStart or i.Position)).Magnitude < 6 and not fired and CC.mobEnabled then ccRunCombo(def.combo) end
        end
        fired = false
    end)
    ccRefs[def.id] = { btn = b, stroke = s }
end

local function ccProbe(url)
    local ok, res = pcall(function() return game:HttpGet(url) end)
    if not ok then return false, "HttpGet failed: "..tostring(res) end
    if type(res) ~= "string" then return false, "HttpGet "..typeof(res) end
    if #res == 0 then return false, "empty" end
    local h = res:sub(1,200):lower()
    if h:find("<!doctype") or h:find("<html") or h:find("<head") then return false, "HTML not Lua:\n\n"..res:sub(1,200) end
    return true, res
end

local function ccLoadCustom()
    if customLoaded then return true end
    local ok, body = ccProbe(customURL)
    if not ok then warn("[TPSULT] "..body); ccShowPopup("Custom Controls: fetch failed", body); return false end
    local ls = loadstring or load
    if type(ls) ~= "function" then ccShowPopup("execute failed", "no loadstring"); return false end
    local f, perr = ls(body)
    if not f then ccShowPopup("compile error", tostring(perr)); return false end
    local ok2, rerr = pcall(f)
    if not ok2 then ccShowPopup("runtime error", tostring(rerr)); return false end
    customLoaded = true; return true
end
