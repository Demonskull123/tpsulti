local function deleteHiddenScripts()
    local deletedCount = 0

    for _, obj in pairs(game:GetDescendants()) do
        if (obj:IsA("Script") or obj:IsA("LocalScript") or obj:IsA("ModuleScript")) then
            for _, name in pairs(scriptsToDelete) do
                if obj.Name == name then
                    pcall(function()
                        obj:Destroy()
                        deletedCount = deletedCount + 1
                        print("Deleted: " .. obj.Name .. " from " .. tostring(obj.Parent))
                    end)
                end
            end
        end
    end

    if getnilinstances then
        for _, obj in pairs(getnilinstances()) do
            if obj:IsA("LuaSourceContainer") then
                for _, name in pairs(scriptsToDelete) do
                    if obj.Name == name then
                        pcall(function()
                            obj:Destroy()
                            deletedCount = deletedCount + 1
                            print("Deleted nil script: " .. obj.Name)
                        end)
                    end
                end
            end
        end
    end

    return deletedCount
end

local function checkAndDestroy(script)
    if script:IsA("LocalScript") or script:IsA("Script") then
        local scriptName = script.Name
        local scriptNameLower = scriptName:lower()

        if scriptName == "" or scriptName:match("^%s*$") then
            warn("Destroying script with empty/whitespace name:", script:GetFullName())
            pcall(function()
                script.Disabled = true
                script:Destroy()
            end)
            return true
        end

        for _, keyword in pairs(blockedKeywords) do
            if scriptNameLower:find(keyword:lower()) then
                warn("Destroying blocked script:", scriptName, "Path:", script:GetFullName())
                pcall(function()
                    script.Disabled = true
                    script:Destroy()
                end)
                return true
            end
        end
    end
    return false
end

local reachSettings = {
    SphereEnabled = false,
    SphereVisualizer = false,
    Radius = 6,
    HeadExcludeMargin = 0.5,
}

local Character = player.Character or player.CharacterAdded:Wait()
local SphereHitbox = nil
local SphereVisualizer = nil
local ActiveTouches = {}
local TOUCH_COOLDOWN = 0.1

local function FindBallPart(part)
    if part:IsA("BasePart") and (part.Name == "TPS" or part.Name == "PSoccerBall") then
        return part
    end
    local current = part
    while current and current ~= workspace do
        if current:IsA("BasePart") and current.Name == "PSoccerBall" then
            return current
        end
        current = current.Parent
    end
    return nil
end

local function DestroySphere()
    if SphereHitbox then SphereHitbox:Destroy() SphereHitbox = nil end
    if SphereVisualizer then SphereVisualizer:Destroy() SphereVisualizer = nil end
    table.clear(ActiveTouches)
end

local function AttachPartFor(c)
    return c:FindFirstChild("UpperTorso")
        or c:FindFirstChild("Torso")
        or c:FindFirstChild("HumanoidRootPart")
end

local function CreateSphere()
    DestroySphere()
    if not Character or not Character.Parent then return end
    if not reachSettings.SphereEnabled then return end

    local attach = AttachPartFor(Character)
    if not attach then return end

    local d = reachSettings.Radius * 2

    local s = Instance.new("Part")
    s.Name        = "ReachSphere"
    s.Shape       = Enum.PartType.Ball
    s.Size        = Vector3.new(d, d, d)
    s.Anchored    = false
    s.CanCollide  = false
    s.CanQuery    = true
    s.CanTouch    = true
    s.Massless    = true
    s.Transparency = 1
    s.CFrame      = attach.CFrame
    s.Parent      = attach

    local w = Instance.new("WeldConstraint")
    w.Part0 = attach
    w.Part1 = s
    w.Parent = s

    SphereHitbox = s

    if reachSettings.SphereVisualizer then
        local v = Instance.new("Part")
        v.Name        = "ReachSphereVisualizer"
        v.Shape       = Enum.PartType.Ball
        v.Size        = s.Size
        v.Anchored    = false
        v.CanCollide  = false
        v.CanQuery    = false
        v.CanTouch    = false
        v.Massless    = true
        v.Transparency = 0.65
        v.Color       = Color3.fromRGB(0, 255, 0)
        v.Material    = Enum.Material.ForceField
        v.CFrame      = s.CFrame
        v.Parent      = s

        local w2 = Instance.new("WeldConstraint")
        w2.Part0 = s
        w2.Part1 = v
        w2.Parent = v

        SphereVisualizer = v
    end
end

local function ResizeSphere()
    if not SphereHitbox then return end
    local d = reachSettings.Radius * 2
    SphereHitbox.Size = Vector3.new(d, d, d)
    if SphereVisualizer then
        SphereVisualizer.Size = Vector3.new(d, d, d)
    end
end

local function SetVisualizer(on)
    if not SphereHitbox then return end
    if on and not SphereVisualizer then
        local v = Instance.new("Part")
        v.Name        = "ReachSphereVisualizer"
        v.Shape       = Enum.PartType.Ball
        v.Size        = SphereHitbox.Size
        v.Anchored    = false
        v.CanCollide  = false
        v.CanQuery    = false
        v.CanTouch    = false
        v.Massless    = true
        v.Transparency = 0.65
        v.Color       = Color3.fromRGB(0, 255, 0)
        v.Material    = Enum.Material.ForceField
        v.CFrame      = SphereHitbox.CFrame
        v.Parent      = SphereHitbox

        local w = Instance.new("WeldConstraint")
        w.Part0 = SphereHitbox
        w.Part1 = v
        w.Parent = v

        SphereVisualizer = v
    elseif not on and SphereVisualizer then
        SphereVisualizer:Destroy()
        SphereVisualizer = nil
    end
end

local function CheckBallCollisions()
    if not Character or not Character.Parent then return end
    if not SphereHitbox or not SphereHitbox.Parent then return end

    local head = Character:FindFirstChild("Head")
    local headY = head and head.Position.Y or math.huge
    local cutoffY = headY - reachSettings.HeadExcludeMargin

    local now = tick()
    local newTouches = {}

    local op = OverlapParams.new()
    op.FilterDescendantsInstances = { Character }
    op.FilterType = Enum.RaycastFilterType.Exclude

    local parts = workspace:GetPartBoundsInBox(
        SphereHitbox.CFrame,
        SphereHitbox.Size,
        op
    )

    for _, p in ipairs(parts) do
        local ball = FindBallPart(p)
        if ball and ball.Position.Y < cutoffY then
            local key = ball:GetFullName()
            newTouches[key] = true
            if not ActiveTouches[key] or (now - ActiveTouches[key]) > TOUCH_COOLDOWN then
                ActiveTouches[key] = now
                pcall(function()
                    firetouchinterest(ball, SphereHitbox, 0)
                    firetouchinterest(ball, SphereHitbox, 1)
                end)
            end
        end
    end

    for k, t in pairs(ActiveTouches) do
        if not newTouches[k] and (now - t) > TOUCH_COOLDOWN then
            ActiveTouches[k] = nil
        end
    end
end

RunService.Heartbeat:Connect(function()
    if not reachSettings.SphereEnabled then return end
    if not SphereHitbox or not SphereHitbox.Parent then
        CreateSphere()
        return
    end
    pcall(CheckBallCollisions)
end)

player.CharacterAdded:Connect(function(c)
    Character = c
    DestroySphere()
    task.wait(1)
    CreateSphere()
end)

task.wait(1)
CreateSphere()

                            local next = trajectoryPoints[nextIndex]
                            local currentPos, nextPos = current[1], next[1]

                            local midpoint = (currentPos + nextPos) * 0.5

                    if aimBeam then aimBeam.Enabled = false end
                end
            else
                humanoid.AutoRotate = originalAutoRotate
                if aimBeam then aimBeam.Enabled = false end
            end
        end)

        player.CharacterAdded:Connect(function(newChar)
            char = newChar; hrp = char:WaitForChild("HumanoidRootPart"); humanoid = char:WaitForChild("Humanoid")
            originalAutoRotate = humanoid.AutoRotate
            if aimBeam then aimBeam.Attachment0.Parent = hrp; aimBeam.Parent = hrp end
        end)

        print("Distance-based aimbot loaded - aims at closest goal")

            if not humanoidRootPart or not humanoidRootPart.Parent or not head then
                updateCharacterReferences()
                if not humanoidRootPart then return end
            end

            lastSaveTime = currentTime
            lastSavedBall = ball
            lastSavedBallTime = currentTime
            local ballHeight = ball.Position.Y
            local headHeight = head.Position.Y
            local rootHeight = humanoidRootPart.Position.Y

            local useHighKeep = false
            if ballHeight >= headHeight then
                useHighKeep = true
            elseif ballHeight < rootHeight then
                useHighKeep = false
            else
                local distToHead = math.abs(ballHeight - headHeight)
                local distToRoot = math.abs(ballHeight - rootHeight)
                useHighKeep = distToHead < distToRoot
            end

            task.spawn(function()
                local success, err = pcall(function()
                    if useHighKeep then
                        require(keep2Module).Keep2()
                    else
                        require(keep1Module).Keep1()
                    end
                end)

                if success then
                    local keepType = useHighKeep and "High" or "Low"
                    statusLabel.Text = "." .. keepType .. " Keep!"
                    statusLabel.TextColor3 = Color3.fromRGB(50, 255, 50)
                else
                    statusLabel.Text = "Keep Failed [PATCHED]"
                    statusLabel.TextColor3 = Color3.fromRGB(255, 50, 50)
                end

                task.wait(1.2)
                if isEnabled then
                    statusLabel.Text = "Waiting..."
                    statusLabel.TextColor3 = Color3.fromRGB(150, 150, 150)
                end
            end)
        end

        local function updateButtonVisual()
            if isEnabled then
                toggleButton.BackgroundColor3 = Color3.fromRGB(50, 200, 50)
                toggleButton.Text = "ON"
                statusLabel.Text = "Monitoring..."
                statusLabel.TextColor3 = Color3.fromRGB(150, 150, 150)
            else
                toggleButton.BackgroundColor3 = Color3.fromRGB(200, 50, 50)
                toggleButton.Text = "OFF"
                statusLabel.Text = "Ready"
                statusLabel.TextColor3 = Color3.fromRGB(150, 150, 150)
            end
        end

        local isToggling = false

        local function toggleAutoSave()
            if isToggling then return end

            isToggling = true

            isEnabled = not isEnabled

            if isEnabled then
                updateBallCache()
                lastSavedBall = nil
            end

            updateButtonVisual()

            isToggling = false
        end

        local buttonPressed = false

        toggleButton.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
                if not buttonPressed then
                    buttonPressed = true
                    toggleAutoSave()
                end
            end
        end)

        toggleButton.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
                buttonPressed = false
            end
        end)

        local lastCheck = 0
        RunService.Heartbeat:Connect(function()
            if not isEnabled then return end

            if tick() - lastCheck < 0.15 then return end
            lastCheck = tick()

            if not character or not character.Parent then
                updateCharacterReferences()
                if not character then return end
            end

            local ball, distance = findClosestMovingBall()

            if ball and distance <= maxDistance then
                executeSave(ball)
            end
        end)

        player.CharacterAdded:Connect(function(newCharacter)
            updateCharacterReferences()
            task.wait()
            updateButtonVisual()
        end)

        updateBallCache()
        updateButtonVisual()
    end
})

local function ccGetModule(tool, which)
    if not tool then return nil end
    local m = tool:FindFirstChild(which)
    if m then return m end
    for _, c in ipairs(tool:GetChildren()) do
        if c.Name:lower() == which:lower() then return c end
    end
    return nil
end

local function ccJump()
    local RS = game:GetService("ReplicatedStorage")
    local mgr = RS:FindFirstChild("Managers")
    local bm  = mgr and mgr:FindFirstChild("BManager")
    local ani = bm and bm:FindFirstChild("Animate")
    local jmp = ani and ani:FindFirstChild("jump")

    if jmp then
        local ok = pcall(function()
            local mod = require(jmp)
            if type(mod) == "function" then
                mod()
            elseif type(mod) == "table" and type(mod.Play) == "function" then
                mod.Play()
            elseif type(mod) == "table" and type(mod.Jump) == "function" then
                mod.Jump()
            end
        end)
        if ok then return true end
    end

    local h = ccHum()
    if h and h.Parent and h.Health > 0 then
        pcall(function()
            h:SetStateEnabled(Enum.HumanoidStateType.Jumping,  true)
            h:SetStateEnabled(Enum.HumanoidStateType.Freefall, true)
            h.Jump = true
            h:ChangeState(Enum.HumanoidStateType.Jumping)
        end)
        return true
    end
    return false
end

local function ccBuildPayload()
    local pout = {}
    for id, ud in pairs(CC.positions) do
        pout[id] = { xs=ud.X.Scale, xo=ud.X.Offset, ys=ud.Y.Scale, yo=ud.Y.Offset }
    end
    return {
        v = 5,
        btn = { size = CC.btn.size, opacity = CC.btn.opacity, locked = CC.btn.locked },
        positions = pout,
    }
end

local function ccApplyPayload(d)
    if type(d) ~= "table" then return end
    if type(d.btn) == "table" then
        if type(d.btn.size)    == "number"  then CC.btn.size    = d.btn.size end
        if type(d.btn.opacity) == "number"  then CC.btn.opacity = d.btn.opacity end
        if type(d.btn.locked)  == "boolean" then CC.btn.locked  = d.btn.locked end
    end
    if type(d.positions) == "table" then
        for id, p in pairs(d.positions) do
            if type(p) == "table" and p.xs and p.xo and p.ys and p.yo then
                CC.positions[id] = UDim2.new(p.xs, p.xo, p.ys, p.yo)
            end
        end
    end
end

local function ccSaveState()
    if not ccCanSave then return end
    pcall(function() writefile(CC_FILE, Http:JSONEncode(ccBuildPayload())) end)
end
CC_API.SaveState = ccSaveState

    if not ex then return end
    local ok, data = pcall(function() return Http:JSONDecode(readfile(CC_FILE)) end)
    if ok then ccApplyPayload(data) end
end

local function ccAddButton(def)
    local pos = CC.positions[def.id] or CC_DEFAULT_POS[def.id] or UDim2.new(0.5, 0, 0.5, 0)
    CC.positions[def.id] = pos

    local btn = Instance.new("TextButton")
    btn.Name                   = def.id
    btn.Size                   = UDim2.new(0, CC.btn.size, 0, CC.btn.size)
    btn.Position               = pos
    btn.BackgroundColor3       = Color3.fromRGB(22, 22, 28)
    btn.BackgroundTransparency = 1 - CC.btn.opacity
    btn.TextColor3             = Color3.fromRGB(0, 255, 130)
    btn.Font                   = Enum.Font.GothamBold
    btn.TextSize               = 14
    btn.Text                   = def.label or def.id
    btn.BorderSizePixel        = 0
    btn.Active                 = true
    btn.Parent                 = ccGui

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 10)
    corner.Parent = btn

    local stroke = Instance.new("UIStroke")
    stroke.Color = CC.btn.locked and Color3.fromRGB(0,255,130) or Color3.fromRGB(255,200,60)
    stroke.Thickness = 1.4
    stroke.Transparency = 0.35
    stroke.Parent = btn

    local dragging, dragStart, startPos, fired = false, nil, nil, false

    btn.InputBegan:Connect(function(input)
        if input.UserInputType ~= Enum.UserInputType.MouseButton1
        and input.UserInputType ~= Enum.UserInputType.Touch then return end
        if input.UserInputState and input.UserInputState ~= Enum.UserInputState.Begin then return end
        if CC.btn.locked then
            if CC.mobEnabled then ccRunCombo(def.combo) end
            fired = true
            return
        end
        dragging = true
        dragStart = input.Position
        startPos = btn.Position
        fired = false
    end)

    btn.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
            local d = input.Position - dragStart
            btn.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)

    btn.InputEnded:Connect(function(input)
        if input.UserInputType ~= Enum.UserInputType.MouseButton1
        and input.UserInputType ~= Enum.UserInputType.Touch then return end
        if dragging then
            dragging = false
            CC.positions[def.id] = btn.Position
            ccSaveState()
            if (input.Position - (dragStart or input.Position)).Magnitude < 6
            and not fired and CC.mobEnabled then
                ccRunCombo(def.combo)
            end
        end
        fired = false
    end)

    ccRefs[def.id] = { btn = btn, stroke = stroke }
end

local function ccProbe(url)
    local ok, res = pcall(function() return game:HttpGet(url) end)
    if not ok then return false, "HttpGet failed: " .. tostring(res) end
    if type(res) ~= "string" then return false, "HttpGet returned " .. typeof(res) end
    if #res == 0 then return false, "empty body" end
    local head = res:sub(1, 200):lower()
    if head:find("<!doctype") or head:find("<html") or head:find("<head") then
        return false, "URL returned HTML not Lua:\n\n" .. res:sub(1, 200)
    end
    return true, res
end

local function ccLoadCustom()
    if customLoaded then return true end
    local ok, body = ccProbe(customURL)
    if not ok then
        warn("[TPSULT] custom controls: " .. body)
        ccShowPopup("Custom Controls: fetch failed", body)
        return false
    end

    local ls = loadstring or load
    if type(ls) ~= "function" then
        ccShowPopup("Custom Controls: execute failed", "no loadstring/load")
        return false
    end

    local fn, perr = ls(body)
    if not fn then
        local msg = "compile error from " .. customURL .. ":\n\n" .. tostring(perr)
        warn("[TPSULT] custom controls: " .. msg)
        ccShowPopup("Custom Controls: execute failed", msg)
        return false
    end

    local ran, rerr = pcall(fn)
    if not ran then
        local msg = "runtime error from " .. customURL .. ":\n\n" .. tostring(rerr)
        warn("[TPSULT] custom controls: " .. msg)
        ccShowPopup("Custom Controls: execute failed", msg)
        return false
    end

    customLoaded = true
    return true
end
